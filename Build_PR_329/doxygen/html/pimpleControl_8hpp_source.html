18// Per-field residual the outer loop is fed each pass: {initResNorm, finalResNorm}.
19// The native control mirrors OpenFOAM pimpleControl::criteriaSatisfied(): the
20// absolute criterion compares the FINAL residual, the relative criterion compares
21// finalResNorm / storedInitial. Feeding both makes both checks available without
22// a Foam::solverPerformance / solverPerformanceDict round-trip.
23using FieldResidual = std::pair<NeoN::scalar, NeoN::scalar>;
24using ResidualMap = std::map<std::string, FieldResidual>;
25
26/* @brief NeoFOAM-native outer PIMPLE loop control (does NOT reuse OpenFOAM's pimpleControl).
27 *
28 * Owns the corr_ / converged_ state machine that drives the outer corrector loop. It mirrors
29 * OpenFOAM pimpleControl::loop() (pimpleControl.C:200-265) and criteriaSatisfied()
30 * (pimpleControl.C:61-126) byte-for-byte on the case side: it reads the SAME
31 * fvSolution.PIMPLE subdict a stock pimpleFoam case uses (nOuterCorrectors,
32 * residualControl{ p{ tolerance; relTol; } U{ ... } }) and applies per-field abs-OR-rel
33 * convergence with no check on the first outer iteration and a two-phase converged_ exit
34 * (an extra final pass after the criteria are met, so *Final factors/tolerances fire).
35 *
36 * Unlike OpenFOAM's pimpleControl it is FED NeoN scalar residuals directly via loop(residuals) —
37 * there is no mesh_.data().solverPerformanceDict() round-trip (rationale: the loop is not
38 * a hot path, so "performant" means lean / decoupled / portable).
39 */
40class PimpleControl
41{
42public:
43
44 // Reads the PIMPLE subdict from fvSolution (nOuterCorrectors, residualControl{}).
45 // A missing / malformed PIMPLE block degrades to nOuterCorrectors == 1 and an empty
46 // residualControl (OpenFOAM defaults) — never throws.
47 explicit PimpleControl(const NeoN::Dictionary& fvSolution);
48
49 // True on the last outer pass: converged_ || (corr_ == nOuterCorr_). Drives *Final
50 // relaxation factors + <field>Final solver subdict selection (pimpleControlI.H:92-95).
51 [[nodiscard]] bool finalIter() const;
52
53 // True on the first outer pass (corr_ == 1) — convergence is never checked here.
54 [[nodiscard]] bool firstIter() const;
55
56 // Call once per outer pass. `residuals` holds the {init,final} residuals from the PREVIOUS
57 // pass. Returns true while another outer corrector should run; false when the loop is done
58 // (count exhausted OR residual criteria satisfied after the extra final pass).
59 bool loop(const ResidualMap& residuals);
60
61 [[nodiscard]] NeoN::label nOuterCorrectors() const { return nOuterCorr_; }
62
63private:
64
65 bool criteriaSatisfied(const ResidualMap& residuals);
