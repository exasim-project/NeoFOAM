25inline void reportContinuityError(
26 const NeoN::finiteVolume::cellCentred::SurfaceField<NeoN::scalar>& phi,
27 const RunTime& rt,
28 NeoN::scalar& cumulative
29)
30{
31 const NeoN::scalar dt = rt.dt;
32 const auto nCells = static_cast<NeoN::localIdx>(rt.nfMesh.nCells());
33
34 NeoN::dsl::Expression<NeoN::scalar> divExpr(rt.exec);
35 divExpr.addOperator(NeoN::dsl::exp::div(phi));
36 NeoN::Vector<NeoN::scalar> divPhi = divExpr.explicitOperation(nCells);
37
38 const auto divPhiView = divPhi.view();
39 const auto volView = rt.nfMesh.cellVolumes().view();
40
41 NeoN::scalar localAbsVolSum = 0.0;
42 NeoN::parallelReduce(
43 rt.exec,
44 {0, static_cast<size_t>(nCells)},
45 NEON_LAMBDA(const size_t i, NeoN::scalar& s) {
46 s += Kokkos::abs(divPhiView[i]) * volView[i];
47 },
48 localAbsVolSum
49 );
50
51 NeoN::scalar localSignedVolSum = 0.0;
52 NeoN::parallelReduce(
53 rt.exec,
54 {0, static_cast<size_t>(nCells)},
55 NEON_LAMBDA(const size_t i, NeoN::scalar& s) { s += divPhiView[i] * volView[i]; },
56 localSignedVolSum
57 );
58
59 NeoN::scalar localTotalVol = 0.0;
60 NeoN::parallelReduce(
61 rt.exec,
62 {0, static_cast<size_t>(nCells)},
63 NEON_LAMBDA(const size_t i, NeoN::scalar& s) { s += volView[i]; },
64 localTotalVol
65 );
66
67 NeoN::scalar globalAbsVolSum = localAbsVolSum;
68 NeoN::scalar globalSignedVolSum = localSignedVolSum;
69 NeoN::scalar totalVol = localTotalVol;
70
71 if (rt.mpiEnvironment.isInitialized() && rt.mpiEnvironment.sizeRank() > 1)
72 {
73 NeoN::mpi::allReduce(globalAbsVolSum, NeoN::mpi::ReduceOp::Sum, rt.mpiEnvironment.comm());
74 NeoN::mpi::allReduce(
75 globalSignedVolSum,
76 NeoN::mpi::ReduceOp::Sum,
77 rt.mpiEnvironment.comm()
78 );
79 NeoN::mpi::allReduce(totalVol, NeoN::mpi::ReduceOp::Sum, rt.mpiEnvironment.comm());
80 }
81
82 const NeoN::scalar sumLocalContErr = dt * globalAbsVolSum / totalVol;
83 const NeoN::scalar globalContErr = dt * globalSignedVolSum / totalVol;
84 cumulative += globalContErr;
85
86 NeoN::Logging::info(
87 "time step continuity errors : sum local = {}, global = {}, cumulative = {}",
88 sumLocalContErr,
89 globalContErr,
90 cumulative
91 );
92}
