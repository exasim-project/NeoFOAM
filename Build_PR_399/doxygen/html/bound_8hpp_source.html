44inline bool bound(
45 NeoN::finiteVolume::cellCentred::VolumeField<NeoN::scalar>& vsf,
46 const NeoN::scalar lowerBound,
47 BoundCache& cache
48)
49{
50 namespace fvcc = NeoN::finiteVolume::cellCentred;
51 using NeoN::localIdx;
52 using NeoN::scalar;
53
54 const auto exec = vsf.exec();
55 const auto& mesh = vsf.mesh();
56 const auto nCells = vsf.internalVector().size();
57
58 auto vsfV = vsf.internalVector().view();
59
60 scalar minVsf = std::numeric_limits<scalar>::max();
61 Kokkos::Min<scalar> minReducer(minVsf);
62 NeoN::parallelReduce(
63 exec,
64 {0, nCells},
65 NEON_LAMBDA(const localIdx i, scalar& m) { m = Kokkos::min(m, vsfV[i]); },
66 minReducer
67 );
68
69 scalar maxVsf = std::numeric_limits<scalar>::lowest();
70 Kokkos::Max<scalar> maxReducer(maxVsf);
71 NeoN::parallelReduce(
72 exec,
73 {0, nCells},
74 NEON_LAMBDA(const localIdx i, scalar& m) { m = Kokkos::max(m, vsfV[i]); },
75 maxReducer
76 );
77
78 scalar sumVsf = 0.0;
79 NeoN::parallelReduce(
80 exec,
81 {0, nCells},
82 NEON_LAMBDA(const localIdx i, scalar& s) { s += vsfV[i]; },
83 sumVsf
84 );
85 scalar cellCount = static_cast<scalar>(nCells);
86
87#ifdef NF_WITH_MPI_SUPPORT
88 if (mesh.boundaryMesh().isDistributed())
89 {
90 NeoN::mpi::Environment env;
91 MPI_Allreduce(MPI_IN_PLACE, &minVsf, 1, NeoN::mpi::getType<scalar>(), MPI_MIN, env.comm());
92 MPI_Allreduce(MPI_IN_PLACE, &maxVsf, 1, NeoN::mpi::getType<scalar>(), MPI_MAX, env.comm());
93 scalar sums[2] = {sumVsf, cellCount};
94 MPI_Allreduce(MPI_IN_PLACE, sums, 2, NeoN::mpi::getType<scalar>(), MPI_SUM, env.comm());
95 sumVsf = sums[0];
96 cellCount = sums[1];
97 }
98#endif
99
100 if (minVsf >= lowerBound)
101 {
102 return false;
103 }
104
105 NeoN::Logging::info(
106 "bounding {}, min: {} max: {} average: {}",
107 vsf.name,
108 minVsf,
109 maxVsf,
110 sumVsf / cellCount
111 );
112
113 // fvc::average(max(vsf, lowerBound)): face-area-weighted cell average of the linearly
114 // interpolated bounded field. Accumulated per face over owner and neighbour, matching
115 // surfaceSum(magSf*ssf)/surfaceSum(magSf) over internal AND boundary faces.
116 NeoN::Vector<scalar> num(exec, nCells, scalar(0));
117 auto numV = num.view();
118
119 const bool buildSumFaceArea =
120 !cache.sumFaceArea.has_value() || cache.sumFaceArea->size() != nCells;
121 if (buildSumFaceArea)
122 {
123 cache.sumFaceArea.emplace(exec, nCells, scalar(0));
124 }
125 auto denV = cache.sumFaceArea->view();
126
127 const auto geo = fvcc::GeometryScheme::readOrCreate(mesh);
128 const auto [wS, areaS, ownerS, neighS] = NeoN::views(
129 geo->weights().internalVector(),
130 mesh.faceAreas(),
131 mesh.faceOwners(),
132 mesh.faceNeighbors()
133 );
134
135 const bool buildSum = buildSumFaceArea;
136 NeoN::parallelFor(
137 exec,
138 {0, mesh.nInternalFaces()},
139 NEON_LAMBDA(const localIdx facei) {
140 const auto own = ownerS[facei];
141 const auto nei = neighS[facei];
142 const scalar bOwn = Kokkos::max(vsfV[own], lowerBound);
143 const scalar bNei = Kokkos::max(vsfV[nei], lowerBound);
144 const scalar faceValue = wS[facei] * bOwn + (scalar(1) - wS[facei]) * bNei;
145 const scalar aSf = areaS[facei];
146 Kokkos::atomic_add(&numV[own], aSf * faceValue);
147 Kokkos::atomic_add(&numV[nei], aSf * faceValue);
148 if (buildSum)
149 {
150 Kokkos::atomic_add(&denV[own], aSf);
151 Kokkos::atomic_add(&denV[nei], aSf);
152 }
153 },
154 "bound::averageInternal"
155 );
156
157 const auto& bMesh = mesh.boundaryMesh();
158 const auto [bOwners, bAreas] = NeoN::views(bMesh.faceOwners(), bMesh.faceAreas());
159 const auto bVsfV = vsf.boundaryData().value().view();
160 const auto nAllBoundaryFaces = vsf.boundaryData().value().size();
161
162 NeoN::parallelFor(
163 exec,
164 {0, nAllBoundaryFaces},
165 NEON_LAMBDA(const localIdx bfi) {
166 const auto own = bOwners[bfi];
167 const scalar aSf = bAreas[bfi];
168 Kokkos::atomic_add(&numV[own], aSf * Kokkos::max(bVsfV[bfi], lowerBound));
169 if (buildSum)
170 {
171 Kokkos::atomic_add(&denV[own], aSf);
172 }
173 },
174 "bound::averageBoundary"
175 );
176
177 // max(max(vsf, average * pos0(-vsf)), lowerBound): the average only replaces cells that are
178 // zero or negative; everything else keeps its value and is merely floored.
179 NeoN::parallelFor(
180 exec,
181 {0, nCells},
182 NEON_LAMBDA(const localIdx celli) {
183 // Floor the divisor rather than branch on it: the compiler if-converts such a branch
184 // into an unconditional division, which would raise FE_INVALID under FOAM_SIGFPE for
185 // a zero face-area sum even though the quotient is discarded.
186 const scalar avg = numV[celli] / Kokkos::max(denV[celli], NeoN::ROOTVSMALL);
187 const scalar refill = vsfV[celli] <= scalar(0) ? avg : scalar(0);
188 vsfV[celli] = Kokkos::max(Kokkos::max(vsfV[celli], refill), lowerBound);
189 },
190 "bound::apply"
191 );
192
193 auto bValueV = vsf.boundaryData().value().view();
194 NeoN::parallelFor(
195 exec,
196 {0, nAllBoundaryFaces},
197 NEON_LAMBDA(const localIdx bfi) { bValueV[bfi] = Kokkos::max(bValueV[bfi], lowerBound); },
198 "bound::applyBoundary"
199 );
200
201 return true;
202}
