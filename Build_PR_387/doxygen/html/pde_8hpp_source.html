149 void setReference(NeoN::localIdx pRefCell, NeoN::scalar pRefValue)
150 {
151 // Accept the call in serial (MPI not initialised — mpiRank is the
152 // sentinel -1, which would compare unequal to 0 if cast to size_t)
153 // and on rank 0 in parallel. SetReference::operator() further guards
154 // the matrix mutation against non-rank-0 in initialised MPI.
155 // When runTime_ is null (1-arg constructor path), default to accepting
156 // on all ranks — SetReference::operator() guards non-rank-0 internally.
157 bool accept = true;
158 if (runTime_ != nullptr)
159 {
160 const auto& mpiEnv = runTime_->mpiEnvironment;
161 accept = !mpiEnv.isInitialized() || mpiEnv.rank() == 0;
162 }
163 if (accept)
164 {
165 needReference_ = true;
166 pRefCell_ = pRefCell;
167 pRefValue_ = pRefValue;
168 }
169 }

240 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
241 {
242 // Assemble and relax the owned ls_ so computeRAUandHByA reads the relaxed diagonal.
243 // Apply -grad p to ls_->rhs() in place (avoids copying the matrix), snapshot the rhs
244 // beforehand and restore it after solve so ls_ retains the H-system for computeRAUandHByA.
245 assemble();
246 relaxOwnedLs();
247
248
249 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
250 rhsExpr.read(NeoFOAM::expandSchemeDefaults(runTime_->fvSchemesDict, rhsExpr, psi_->name));
251 auto savedRhs = NeoN::Vector<ValueType>(ls_->rhs());
252 rhsExpr.assembleExplicitSource(*ls_, psi_->mesh());
253
254 auto solverDict = runTime_->fvSolutionDict.subDict("solvers");
255 const std::string finalKey = psi_->name + "Final";
256 auto fvSolution = (finalIter_ && solverDict.isDict(finalKey))
257 ? solverDict.subDict(finalKey)
258 : solverDict.subDict(psi_->name);
259 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
260 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
261 stripNeoFOAMKeys(fvSolution);
262 auto solver = NeoN::la::Solver(psi_->exec(), fvSolution);
263 // Do some sanity checks before trying to solve
264 // NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
265 auto stats = solver.solve(*ls_, psi_->internalVector());
266
267 ls_->rhs() = savedRhs;
268
269 reportSolverStats(stats, fvSolution);
270 return stats;
271 }

274 NeoN::la::SolverStats solveWith(NeoN::la::Solver& solver, VolumeField& psi, RunTime& rt)
275 {
276 initIfNeeded(psi, rt);
277
278 expr_.read(NeoFOAM::expandSchemeDefaults(rt.fvSchemesDict, expr_, psi.name));
279 ls_->reset();
280 expr_.assemble(rt.t, rt.dt, *ls_, psi.mesh());
281
282 const auto alpha =
283 lookupEqnRelaxation(rt.fvSolutionDict, psi.name, finalIter_).value_or(1.0);
284 NeoN::dsl::applyMatrixRelaxation(*ls_, psi, alpha);
285
286 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
287 {
288 if (needReference_)
289 {
290 NeoN::dsl::SetReference<ValueType> refFunct(pRefCell_, pRefValue_);
291 refFunct(*ls_);
292 }
293 }
294
295 auto solverDict = rt.fvSolutionDict.subDict("solvers");
296 const std::string finalKey = psi.name + "Final";
297 const bool useFinal = finalIter_ && solverDict.isDict(finalKey);
298 auto fieldSolverDict =
299 useFinal ? solverDict.subDict(finalKey) : solverDict.subDict(psi.name);
300 stripNeoFOAMKeys(fieldSolverDict);
301 NeoN::fence(psi.exec());
302 NF_ASSERT(ls_->exec() == psi.exec(), "Executors are not the same");
303 NeoN::la::SolverStats stats;
304 if (useFinal)
305 {
306 NeoN::la::Solver finalSolver(psi.exec(), fieldSolverDict);
307 stats = finalSolver.solve(*ls_, psi.internalVector());
308 }
309 else
310 {
311 stats = solver.solve(*ls_, psi.internalVector());
312 }
313
314 reportSolverStats(stats, fieldSolverDict);
315 return stats;
316 }

367 // Public because NVCC forbids extended __host__ __device__ lambdas
368 // (NEON_LAMBDA) inside private or protected member functions.
369 NeoN::la::SolverStats solveImpl(dsl::Expression<ValueType>& expr, LinearSystem& ls)
370 {
371 // Re-read schemes (idempotent with the constructor read)
372 expr.read(NeoFOAM::expandSchemeDefaults(runTime_->fvSchemesDict, expr, psi_->name));
373
374 // Assemble without post-assembly functors; we apply SetReference separately below
375 // to ensure correct polymorphic dispatch — storing PostAssemblyBase by value causes
376 // object slicing that silently disables virtual overrides.
377 expr.assemble(runTime_->t, runTime_->dt, ls, psi_->mesh());
378
379 // Relaxation MUST precede SetReference: SetReference doubles the ref-cell diagonal,
380 // and relaxing afterwards would corrupt the pin.
381 const auto alpha =
382 lookupEqnRelaxation(runTime_->fvSolutionDict, psi_->name, finalIter_).value_or(1.0);
383 NeoN::dsl::applyMatrixRelaxation(ls, *psi_, alpha);
384
385 // Apply reference-cell pinning (kernel compiled only in pde.cpp via detail helper)
386 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
387 {
388 if (needReference_)
389 {
390 detail::applySetReference(ls, pRefCell_, pRefValue_);
391 }
392 }
393
394 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
395 {
396 if (constraintMask_ != nullptr)
397 {
398 const NeoN::localIdx nCells = psi_->mesh().nCells();
399 detail::applyFixedValueConstraints(
400 ls,
401 constraintMask_->view(),
402 constraintValues_->view(),
403 nCells
404 );
405 }
406 }
407
408 auto solverDict = runTime_->fvSolutionDict.subDict("solvers");
409 const std::string finalKey = psi_->name + "Final";
410 auto fieldSolverDict = (finalIter_ && solverDict.isDict(finalKey))
411 ? solverDict.subDict(finalKey)
412 : solverDict.subDict(psi_->name);
413 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
414 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
415 stripNeoFOAMKeys(fieldSolverDict);
416 NeoN::fence(psi_->exec());
417 NF_ASSERT(ls.exec() == psi_->exec(), "Executors are not the same");
418
419 auto solver = NeoN::la::Solver(psi_->exec(), fieldSolverDict);
420 auto stats = solver.solve(ls, psi_->internalVector());
421
422 reportSolverStats(stats, fieldSolverDict);
423
424 return stats;
425 }
426
427 // Relax the owned ls_ in place. Shared by solve(rhs) and assembleAndRelax() so both
428 // paths use the same alpha lookup. Must run before SetReference (see solveImpl).
429 // Only calls the free function applyMatrixRelaxation — no NEON_LAMBDA in this body.
430 void relaxOwnedLs()
431 {
432 const auto alpha =
433 lookupEqnRelaxation(runTime_->fvSolutionDict, psi_->name, finalIter_).value_or(1.0);
434 NeoN::dsl::applyMatrixRelaxation(*ls_, *psi_, alpha);
435 }

437 // Remove NeoFOAM-specific control keys from a per-field solver dict before it
438 // is passed to NeoN/Ginkgo. Ginkgo's config parser is strict and aborts on any
439 // unrecognised key, so these must be popped here rather than left in place.
440 static void stripNeoFOAMKeys(NeoN::Dictionary& dict)
441 {
442 for (const auto& key : {"assemblyStrategy", "optimize"})
443 {
444 if (dict.contains(key))
445 {
446 dict.remove(key);
447 }
448 }

502 // Per-component name (Ux/Uy/Uz) when a vector field is solved as separate
503 // component systems; the plain field name otherwise (scalar or coupled solve).
504 static std::string componentName(const std::string& base, std::size_t i, std::size_t n)
505 {
506 if (n <= 1) return base;
507 constexpr const char* suffix[3] = {"x", "y", "z"};
508 return i < 3 ? base + suffix[i] : base + "[" + std::to_string(i) + "]";
509 }

511 // OpenFOAM-style per-solve residual report: "<precond><solver>: Solving for
512 // <field>, Initial residual = ..., Final residual = ..., No Iterations N".
513 // The "<precond><solver>" label (e.g. DICPCG) is read from the solver dict's
514 // reportName meta key stashed by mapFvSolution -> no recomputation per solve.
515 // Logging is rank-aware and skips formatting on muted ranks (NeoN shouldLog).
516 void reportSolverStats(
517 const NeoN::la::SolverStats& stats,
518 const NeoN::Dictionary& fieldSolverDict
519 ) const
520 {
521 const std::string label = fieldSolverDict.contains("reportName")
522 ? fieldSolverDict.get<std::string>("reportName")
523 : std::string("Ginkgo");
524 const std::size_t n = stats.entries.size();
525 for (std::size_t i = 0; i < n; ++i)
526 {
527 const auto& stat = stats.entries[i];
528 NeoN::Logging::info(
529 "{}: Solving for {}, Initial residual = {}, Final residual = {}, No Iterations {}",
530 label,
531 componentName(psi_->name, i, n),
532 stat.initResNorm,
533 stat.finalResNorm,
534 stat.numIter
535 );
536 }
537 }
