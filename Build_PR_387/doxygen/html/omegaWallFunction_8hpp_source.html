46inline void setOmegaWallFunction(
47 Field<scalar>& omega,
48 const fvcc::VolumeField<scalar>& k,
49 const fvcc::VolumeField<scalar>& nu,
50 const fvcc::VolumeField<scalar>& nearWallDist,
51 const UnstructuredMesh& mesh,
52 std::pair<localIdx, localIdx> range,
53 scalar beta1,
54 scalar cmu,
55 scalar kappa
56)
57{
58 const scalar cmu25 = Kokkos::pow(cmu, scalar(0.25));
59 // Function-local copy: a namespace-scope constexpr cannot be referenced inside a
60 // __host__ __device__ lambda (it is "undefined in device code"); bind it here first.
61 const scalar omegaMax = OMEGA_WF_OMEGA_MAX;
62
63 auto kInternal = k.internalVector().view();
64 const auto nuBoundary = nu.boundaryData().value().view();
65 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
66
67 auto [refGrad, value, valueFraction, refValue, faceOwners] = views(
68 omega.boundaryData().refGrad(),
69 omega.boundaryData().value(),
70 omega.boundaryData().valueFraction(),
71 omega.boundaryData().refValue(),
72 mesh.boundaryMesh().faceOwners()
73 );
74
75 NeoN::parallelFor(
76 omega.exec(),
77 range,
78 NEON_LAMBDA(const localIdx i) {
79 const localIdx owner = faceOwners[i];
80 const scalar y = nearWallBoundary[i];
81 const scalar nuw = nuBoundary[i];
82 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
83
84 // Viscous sublayer: 6ν / (β₁·y²) (omega.C:218-224)
85 const scalar wVis = 6.0 * nuw / (beta1 * y * y);
86
87 // Log layer: √k / (C_µ^0.25·κ·y) (omega.C:226-234)
88 const scalar wLog = Kokkos::sqrt(kw) / (cmu25 * kappa * y);
89
90 // BINOMIAL blender, n = 2 (upstream default at omega.C:396) —
91 // closed form: ω = √(ωᵥᵢₛ² + ωₗₒg²). Clamped to OMEGA_WF_OMEGA_MAX so this face
92 // value matches the kOmegaSST cell pin exactly (both use the same clamp).
93 const scalar wOmega = Kokkos::min(Kokkos::sqrt(wVis * wVis + wLog * wLog), omegaMax);
94
95 // Set the wall face value only. Upstream also writes
96 // omega_internal[wall_cell] via manipulateMatrix(setValues) — a
97 // matrix-row pin we have no equivalent for. Without that pin a
98 // direct stomp on omega.internalVector() only persists until the
99 // next omega solve overwrites it; meanwhile correctNutInternal
100 // reads the stomped value and collapses nut.internal at the wall
101 // cell. Leaving the internal value alone keeps the fixedValue
102 // semantics correct via the boundary face.
103 value[i] = wOmega;
104 refValue[i] = wOmega;
105 valueFraction[i] = 1.0;
106 refGrad[i] = 0.0;
107 },
108 "setOmegaWallFunction"
109 );
110}
