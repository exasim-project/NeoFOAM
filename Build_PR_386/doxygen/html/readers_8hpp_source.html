35inline NeoN::scalar tokenAsScalar(NeoN::TokenList& tokenList, std::size_t idx)
36{
37 auto& tokens = tokenList.tokens();
38 if (const NeoN::scalar* asScalar = std::any_cast<NeoN::scalar>(&tokens[idx]))
39 {
40 return *asScalar;
41 }
42 // Not stored as a scalar — it was parsed as an integer label. Promote it.
43 return NeoN::scalar(tokenList.get<Foam::label>(idx));
44}

54struct PatchNormal
55{
56 NeoN::Vec3 meanNormal;
57 NeoN::scalar deviation;
58};

60inline PatchNormal meanUnitNormal(const Foam::fvPatch& patch)
61{
62 const Foam::vectorField faceNormals(patch.nf());
63 if (faceNormals.empty())
64 {
65 return {NeoN::Vec3(0.0, 0.0, 0.0), 0.0};
66 }
67
68 Foam::vector mean(Foam::Zero);
69 for (const Foam::vector& faceNormal : faceNormals)
70 {
71 mean += faceNormal;
72 }
73 // Re-normalise: averaging unit normals shortens the sum on a curved patch, and on a
74 // closed one it cancels to (almost) zero — VSMALL keeps that finite, and the deviation
75 // it produces then trips the caller's non-planar warning.
76 mean /= Foam::mag(mean) + Foam::VSMALL;
77
78 NeoN::scalar deviation = 0.0;
79 for (const Foam::vector& faceNormal : faceNormals)
80 {
81 deviation = std::max(deviation, NeoN::scalar(Foam::mag(faceNormal - mean)));
82 }
83 return {convert(mean), deviation};
84}

93template<typename ValueType>
94void insertFrozenTotalPressure(NeoN::Dictionary& dict)
95{
96 if constexpr (std::is_same<ValueType, NeoN::scalar>::value)
97 {
98 if (dict.contains("value"))
99 {

128template<typename ValueType>
129void insertUniformValue(
130 NeoN::Dictionary& dict,
131 const std::string& sourceKey,
132 const std::string& targetKey,
133 const std::string& patchName,
134 const Foam::fvPatch* patch
135)
136{
137 NeoN::TokenList tokenList = dict.get<NeoN::TokenList>(sourceKey);
138 constexpr NeoN::localIdx required = std::is_same<ValueType, NeoN::Vec3>::value ? 4 : 2;
139 if (tokenList.size() < required)
140 {
141 if (patch != nullptr && patch->size() == 0)
142 {
143 dict.insert(targetKey, NeoN::zero<ValueType>());
144 return;
145 }
146 throw std::runtime_error(
147 "patch '" + patchName + "': `" + sourceKey + "` is not a uniform value ("
148 + std::to_string(tokenList.size()) + " tokens, expected " + std::to_string(required)
149 + ")."
150 );
151 }
152 if constexpr (std::is_same<ValueType, NeoN::Vec3>::value)
153 {
154 NeoN::Vec3 value {};
155 value[0] = tokenAsScalar(tokenList, 1);
156 value[1] = tokenAsScalar(tokenList, 2);
157 value[2] = tokenAsScalar(tokenList, 3);
158 dict.insert(targetKey, value);
159 }
160 else
161 {
162 dict.insert(targetKey, tokenAsScalar(tokenList, 1));
163 }
164}

181template<typename FoamType>
182auto readVolBoundaryConditions(const NeoN::UnstructuredMesh& nfMesh, const FoamType& ofVolField)
183{
184 using type_container_t = typename TypeMap<FoamType>::container_type;
185 using type_primitive_t = typename TypeMap<FoamType>::mapped_type;
186
187 // get boundary as dictionary
188 Foam::OStringStream os;
189 ofVolField.boundaryField().writeEntries(os);
190 Foam::IStringStream is(os.str());
191 Foam::dictionary bDict(is);
192
193 // The inserters receive only the patch dictionary, but the translations below need the
194 // patch itself — its geometry (surfaceNormalFixedValue) and its name (the approximation
195 // notices). applyVolInserter points these at the patch it is about to translate. The
196 // notices stream the name as a `const char*`: OpenFOAM's Ostream writes a std::string
197 // as a quoted string token.
198 std::string activePatchName;
199 const Foam::fvPatch* activeFoamPatch = nullptr;
200
201 std::map<std::string, std::function<void(NeoN::Dictionary&)>>
202 patchInserter {
203 {"fixedGradient",
204 [](auto& dict)
205 {
206 dict.insert("type", std::string("fixedGradient"));
207 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("value");
208 type_primitive_t fixedGradient = tokenList.get<type_primitive_t>(1);
209 dict.insert("fixedGradient", fixedGradient);
210 }},
211 {"zeroGradient",
212 [&](auto& dict)
213 {
214 dict.insert("type", std::string("fixedGradient"));
215 dict.insert("fixedGradient", NeoN::zero<type_primitive_t>());
216 }},
217 {"fixedValue",
218 [](auto& dict)
219 {
220 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("value");
221 auto fixedValue = NeoN::zero<type_primitive_t>();
222
223 // NOTE FIXME in parallel cases we end up with token.size ==1
224 // leading to nonuniform as first token. This means probably that
225 // parsing the foam dictionary aborts early and omits 0();
226 if (tokenList.size() > 1)
227 {
228 dict.insert("type", std::string("fixedValue"));
229 // OpenFOAM classifies each numeric token independently: a literal
230 // with a decimal point is a SCALAR, a bare integer is a LABEL. A
231 // single vector value can therefore mix the two (e.g. `(0.1 0 0)`
232 // -> [scalar, label, label]). Read every component via
233 // detail::tokenAsScalar so each is promoted to scalar on its own,
234 // instead of choosing one branch for the whole value based solely
235 // on the first component (which caused bad_any_cast on index 2/3).
236 if constexpr (std::is_same<type_primitive_t, NeoN::Vec3>::value)
237 {
238 NeoN::Vec3 tmpFixedValue {};
239 tmpFixedValue[0] = detail::tokenAsScalar(tokenList, 1);
240 tmpFixedValue[1] = detail::tokenAsScalar(tokenList, 2);
241 tmpFixedValue[2] = detail::tokenAsScalar(tokenList, 3);
242 dict.insert("fixedValue", tmpFixedValue);
243 return;
244 }
245 else
246 {
247 fixedValue = detail::tokenAsScalar(tokenList, 1);
248 }
249 dict.insert("fixedValue", fixedValue);
250 }
251 else
252 {
253 // FIXME is this an empty boundary?
254 dict.insert("type", std::string("empty"));
255 // left blank
256 }
257 }},
258 {"uniformFixedValue",
259 [](auto& dict)
260 {
261 // uniformFixedValue stores its value as a Function1. We only support
262 // the (by far most common) `constant <value>` form, which converts to
263 // a TokenList of [word "constant", value...]; the value therefore
264 // starts at index 1, exactly like the `value uniform ...` case above.
265 dict.insert("type", std::string("fixedValue"));
266 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("uniformValue");
267 if constexpr (std::is_same<type_primitive_t, NeoN::Vec3>::value)
268 {
269 NeoN::Vec3 tmpFixedValue {};
270 tmpFixedValue[0] = detail::tokenAsScalar(tokenList, 1);
271 tmpFixedValue[1] = detail::tokenAsScalar(tokenList, 2);
272 tmpFixedValue[2] = detail::tokenAsScalar(tokenList, 3);
273 dict.insert("fixedValue", tmpFixedValue);
274 }
275 else
276 {
277 dict.insert("fixedValue", detail::tokenAsScalar(tokenList, 1));
278 }
279 }},
280 {"noSlip", // TODO specialize for vector
281 [](auto& dict)
282 {
283 dict.insert("type", std::string("fixedValue"));
284 dict.insert("fixedValue", type_primitive_t {});
285 }},
286 {"calculated", [](auto& dict) { dict.insert("type", std::string("calculated")); }},
287 {"processor", [](auto& dict) { dict.insert("type", std::string("processor")); }},
288 {"extrapolatedCalculated",
289 [](auto& dict) { dict.insert("type", std::string("calculated")); }},
290 {"empty", [](auto& dict) { dict.insert("type", std::string("empty")); }},
291 {"symmetryPlane", [](auto& dict) { dict.insert("type", std::string("symmetry")); }},
292 {"symmetry", [](auto& dict) { dict.insert("type", std::string("symmetry")); }},
293 {"nutUSpaldingWallFunction",
294 [](auto& dict) { dict.insert("type", std::string("nutUSpaldingWallFunction")); }},
295 {"kqRWallFunction",
296 [](auto& dict) { dict.insert("type", std::string("kqRWallFunction")); }},
297 {"omegaWallFunction",
298 [](auto& dict) { dict.insert("type", std::string("omegaWallFunction")); }},
299 {"epsilonWallFunction",
300 [](auto& dict) { dict.insert("type", std::string("epsilonWallFunction")); }},
301 {"nutkWallFunction",
302 [](auto& dict) { dict.insert("type", std::string("nutkWallFunction")); }},
303 {"inletOutlet",
304 [](auto& dict)
305 {
306 dict.insert("type", std::string("inletOutlet"));
307 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("inletValue");
308 if constexpr (std::is_same<type_primitive_t, NeoN::Vec3>::value)
309 {
310 NeoN::Vec3 inletValue {};
311 if (tokenList.size() >= 4)
312 {
313 inletValue[0] = detail::tokenAsScalar(tokenList, 1);
314 inletValue[1] = detail::tokenAsScalar(tokenList, 2);
315 inletValue[2] = detail::tokenAsScalar(tokenList, 3);
316 }
317 dict.insert("inletValue", inletValue);
318 }
319 else
320 {
321 dict.insert(
322 "inletValue",
323 tokenList.size() >= 2 ? detail::tokenAsScalar(tokenList, 1)
324 : type_primitive_t {}
325 );
326 }
327 }},
328 {"freestream",
329 [&](auto& dict)
330 {
331 WarningInFunction
332 << "freestream on patch '" << activePatchName.c_str()
333 << "' is approximated as a fixedValue at the freestreamValue.\n"
334 " OpenFOAM blends it with zeroGradient using valueFraction = neg(phi);"
335 " this pins the valueFraction = 1 endpoint, so outflow faces stay Dirichlet"
336 " instead of extrapolating."
337 << Foam::endl;
338 // OpenFOAM's freestream<Type> derives from inletOutlet<Type> and only renames
339 // inletValue to freestreamValue, so inletOutlet looks like the exact mapping --
340 // but NeoN's inletOutlet consults the flux only in its
341 // correctBoundaryCondition(field, BoundaryContext) overload, and this path calls
342 // the no-context one, which treats every face as outflow. On airFoil2D that left
343 // the far-field nuTilda unpinned and it grew from its 4e-05 freestreamValue to
344 // ~41 against a native peak of 0.29. Same reasoning as freestreamVelocity below:
345 // pin the value rather than lose the far-field datum. Restore the inletOutlet
346 // mapping once a phi BoundaryContext reaches correctBoundaryCondition.
347 dict.insert("type", std::string("fixedValue"));
348 detail::insertUniformValue<type_primitive_t>(
349 dict,
350 "freestreamValue",
351 "fixedValue",
352 activePatchName,
353 activeFoamPatch
354 );
355 }},
356 {"freestreamVelocity",
357 [&](auto& dict)
358 {
359 WarningInFunction
360 << "freestreamVelocity on patch '" << activePatchName.c_str()
361 << "' is approximated as a fixedValue at the freestreamValue.\n"
362 " OpenFOAM blends the freestream value with zeroGradient using"
363 " valueFraction = neg(phi); this pins the valueFraction = 1 endpoint, so"
364 " outflow faces stay Dirichlet instead of extrapolating."
365 << Foam::endl;
366 // Deliberately not inletOutlet: NeoN's inletOutlet only consults the flux in its
367 // correctBoundaryCondition(field, BoundaryContext) overload, and the
368 // incompressibleFluidNeoN path calls the no-context one, which treats every face
369 // as outflow. The whole far field would go Neumann for U, and airFoil2D has no
370 // other inlet — the freestream forcing would vanish silently and the run would
371 // produce garbage instead of failing. fixedValue keeps the momentum system
372 // well posed.
373 dict.insert("type", std::string("fixedValue"));
374 detail::insertUniformValue<type_primitive_t>(
375 dict,
376 "freestreamValue",
377 "fixedValue",
378 activePatchName,
379 activeFoamPatch
380 );
381 }},
382 {"freestreamPressure",
383 [&](auto& dict)
384 {
385 WarningInFunction
386 << "freestreamPressure on patch '" << activePatchName.c_str()
387 << "' is approximated as zeroGradient.\n"
388 " NeoN models neither the flux-driven blend nor the freestreamValue"
389 " datum on this patch, so the far-field pressure follows the interior."
390 << Foam::endl;
391 // The complement of the freestreamVelocity mapping above: U is Dirichlet on the
392 // far field, so p has to be Neumann there for the pressure system to stay
393 // well posed.
394 dict.insert("type", std::string("fixedGradient"));
395 dict.insert("fixedGradient", NeoN::zero<type_primitive_t>());
396 }},
397 // Two-phase VoF outlet/pressure conditions from the damBreak case. These
398 // are outflow-dominant and are approximated here as zeroGradient so the
399 // fields can be read; a faithful treatment lands with the VoF solve.
400 {"pressureInletOutletVelocity",
401 [&](auto& dict)
402 {
403 dict.insert("type", std::string("fixedGradient"));
404 dict.insert("fixedGradient", NeoN::zero<type_primitive_t>());
405 }},
406 {"fixedFluxPressure",
407 [&](auto& dict)
408 {
409 // Faithful wall fixedFluxPressure: the per-face refGrad is set externally by
410 // NeoFOAM::constrainPressure so the projection cancels the buoyancy/capillary
411 // wall face flux. The FixedFluxPressure BC ignores the dict (refGrad is
412 // zero-initialised -> zeroGradient until the first constrainPressure).
413 dict.insert("type", std::string("fixedFluxPressure"));
414 }},
415 {"totalPressure",
416 [&](auto& dict)
417 {
418 // p_rgh atmosphere: pin the pressure datum with a fixedValue at the stored
419 // patch value so the p_rgh solve is well-posed. The dynamic-head correction
420 // is negligible for the gravity-driven damBreak first version and is deferred
421 // with surface tension. Non-scalar (or valueless) falls back to zeroGradient.
422 detail::insertFrozenTotalPressure<type_primitive_t>(dict);
423 }},
424 {"uniformTotalPressure",
425 [&](auto& dict)
426 {
427 WarningInFunction
428 << "uniformTotalPressure on patch '" << activePatchName.c_str()
429 << "' is approximated as a fixedValue at the total pressure OpenFOAM"
430 " evaluated for the start time.\n NeoN applies neither the p0(t)"
431 " Function1 nor the dynamic-head correction p0 - 0.5|U|^2, so the patch"
432 " pressure stays constant for the whole run."
433 << Foam::endl;
434 detail::insertFrozenTotalPressure<type_primitive_t>(dict);
435 }},
436 {"slip", [](auto& dict) { dict.insert("type", std::string("slip")); }},
437 {"movingWallVelocity",
438 [&](auto& dict)
439 {
440 WarningInFunction
441 << "movingWallVelocity on patch '" << activePatchName.c_str()
442 << "' is approximated as a stationary no-slip wall, fixedValue (0 0 0).\n"
443 " NeoN has no mesh motion, so the wall velocity U_wall = U_mesh = 0;"
444 " the result differs from OpenFOAM wherever the mesh actually moves."
445 << Foam::endl;
446 dict.insert("type", std::string("fixedValue"));
447 dict.insert("fixedValue", NeoN::zero<type_primitive_t>());
448 }},
449 {"surfaceNormalFixedValue",
450 [&](auto& dict)
451 {
452 // OpenFOAM evaluates refValue*n_f per face; NeoN's fixedValue holds one value
453 // for the whole patch, so refValue is projected onto the patch's mean unit
454 // normal here — exact on a planar patch (every intake in the sweep), an
455 // approximation on a curved one, which is why the deviation is checked.
456 if constexpr (std::is_same<type_primitive_t, NeoN::Vec3>::value)
457 {
458 if (activeFoamPatch == nullptr)
459 {
460 throw std::runtime_error(
461 "surfaceNormalFixedValue on patch '" + activePatchName
462 + "': patch geometry not found in the mesh."
463 );
464 }
465 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("refValue");
466 const std::string* form =
467 tokenList.size() > 1 ? std::any_cast<std::string>(&tokenList.tokens()[0])
468 : nullptr;
469 if (form == nullptr || *form != "uniform")
470 {
471 throw std::runtime_error(
472 "surfaceNormalFixedValue on patch '" + activePatchName
473 + "': only a uniform refValue is supported."
474 );
475 }
476 const NeoN::scalar refValue = detail::tokenAsScalar(tokenList, 1);
477 const auto [meanNormal, deviation] = detail::meanUnitNormal(*activeFoamPatch);
478
479 if (dict.contains("ramp"))
480 {
481 WarningInFunction
482 << "surfaceNormalFixedValue on patch '" << activePatchName.c_str()
483 << "' carries a `ramp` Function1, which NeoN does not model: the full"
484 " refValue is applied from the first time step."
485 << Foam::endl;
486 }
487 if (deviation > 1e-6)
488 {
489 WarningInFunction
490 << "surfaceNormalFixedValue on patch '" << activePatchName.c_str()
491 << "' is not planar (face normals deviate by up to " << deviation
492 << " from the patch mean): the single fixedValue refValue*n uses the"
493 " mean normal for every face."
494 << Foam::endl;
495 }
496 dict.insert("type", std::string("fixedValue"));
497 dict.insert("fixedValue", refValue * meanNormal);
498 }
499 else
500 {
501 throw std::runtime_error(
502 "surfaceNormalFixedValue on patch '" + activePatchName
503 + "' is only defined for vector fields."
504 );
505 }
506 }}
507 };

509 auto applyVolInserter =
510 [&](const std::string& patchName, const std::string& bcType, NeoN::Dictionary& dict)
511 {
512 auto it = patchInserter.find(bcType);
513 if (it == patchInserter.end())
514 {
515 std::string supported;
516 for (const auto& [key, _] : patchInserter)
517 supported += "\n " + key;
518 throw std::runtime_error(
519 "Unsupported boundary condition type '" + bcType + "' on patch '" + patchName
520 + "'.\nSupported types:" + supported
521 );
522 }
523 activePatchName = patchName;
524 const Foam::label foamPatchID = ofVolField.mesh().boundaryMesh().findPatchID(patchName);
525 activeFoamPatch = foamPatchID >= 0 ? &ofVolField.mesh().boundary()[foamPatchID] : nullptr;
526 it->second(dict);
527 };
528
529 int patchi = 0;
530 std::vector<fvcc::VolumeBoundary<type_primitive_t>> bcs;
531 // do non processor first
532 for (const auto bName : bDict.toc())
533 {
534 Foam::dictionary patchDict = bDict.subDict(bName);
535 std::string bcType = patchDict.get<Foam::word>("type");
536 if (bcType != "processor")
537 {
538 NeoN::Dictionary neoPatchDict = convert(patchDict);
539 applyVolInserter(bName, bcType, neoPatchDict);
540 bcs.emplace_back(nfMesh, neoPatchDict, patchi);
541 patchi++;
542 }
543 }
544 for (const auto bName : bDict.toc())
545 {
546 Foam::dictionary patchDict = bDict.subDict(bName);
547 std::string bcType = patchDict.get<Foam::word>("type");
548 if (bcType == "processor")
549 {
550 NeoN::Dictionary neoPatchDict = convert(patchDict);
551 applyVolInserter(bName, bcType, neoPatchDict);
552 bcs.emplace_back(nfMesh, neoPatchDict, patchi);
553 patchi++;
554 }
555 }
556 return bcs;
557}

559template<typename FoamType>
560auto readSurfaceBoundaryConditions(
561 const NeoN::UnstructuredMesh& uMesh,
562 const FoamType& surfaceField
563)
564{
565 using type_container_t = typename TypeMap<FoamType>::container_type;
566 using type_primitive_t = typename TypeMap<FoamType>::mapped_type;
567
568 std::vector<fvcc::SurfaceBoundary<type_primitive_t>> bcs;
569
570 // get boundary as dictionary
571 Foam::OStringStream os;
572 surfaceField.boundaryField().writeEntries(os);
573 Foam::IStringStream is(os.str());
574 Foam::dictionary bDict(is);
575 int patchi = 0;
576
577 // TODO this approach fails for procBoundary0to1
578 std::map<std::string, std::function<void(NeoN::Dictionary&)>> patchInserter {
579 {"fixedGradient", [](auto& dict) { dict.insert("type", std::string("fixedGradient")); }},
580 {"zeroGradient",
581 [&](auto& dict)
582 {
583 dict.insert("type", std::string("fixedGradient"));
584 dict.insert("fixedGradient", type_primitive_t {});
585 }},
586 {"fixedValue",
587 [](auto& dict)
588 {
589 dict.insert("type", std::string("fixedValue"));
590 dict.insert("fixedValue", type_primitive_t {});
591 }},
592 {"noSlip", // TODO specialize for vector
593 [](auto& dict)
594 {
595 dict.insert("type", std::string("fixedValue"));
596 dict.insert("fixedValue", type_primitive_t {});
597 }},
598 {"calculated", [](auto& dict) { dict.insert("type", std::string("calculated")); }},
599 {"processor", [](auto& dict) { dict.insert("type", std::string("processor")); }},
600 {"empty", [](auto& dict) { dict.insert("type", std::string("empty")); }},
601 {"symmetryPlane", [](auto& dict) { dict.insert("type", std::string("symmetry")); }},
602 {"symmetry", [](auto& dict) { dict.insert("type", std::string("symmetry")); }}
603 };
604
605 auto applySurfaceInserter =
606 [&](const std::string& patchName, const std::string& bcType, NeoN::Dictionary& dict)
607 {
608 auto it = patchInserter.find(bcType);
609 if (it == patchInserter.end())
610 {
611 std::string supported;
612 for (const auto& [key, _] : patchInserter)
613 supported += "\n " + key;
614 throw std::runtime_error(
615 "Unsupported boundary condition type '" + bcType + "' on patch '" + patchName
616 + "'.\nSupported types:" + supported
617 );
618 }
619 it->second(dict);
620 };
621
622 for (const auto& bName : bDict.toc())
623 {
624 Foam::dictionary patchDict = bDict.subDict(bName);
625 std::string bcType = patchDict.get<Foam::word>("type");
626 if (bcType != "processor")
627 {
628 NeoN::Dictionary neoPatchDict;
629 applySurfaceInserter(bName, bcType, neoPatchDict);
630 bcs.push_back(fvcc::SurfaceBoundary<type_primitive_t>(uMesh, neoPatchDict, patchi));
631 patchi++;
632 }
633 }
634 for (const auto& bName : bDict.toc())
635 {
636 Foam::dictionary patchDict = bDict.subDict(bName);
637 std::string bcType = patchDict.get<Foam::word>("type");
638 if (bcType == "processor")
639 {
640 NeoN::Dictionary neoPatchDict;
641 applySurfaceInserter(bName, bcType, neoPatchDict);
642 bcs.push_back(fvcc::SurfaceBoundary<type_primitive_t>(uMesh, neoPatchDict, patchi));
643 patchi++;
644 }
645 }
646 return bcs;
647}

649template<class FoamFieldType>
650auto constructFrom(
651 const NeoN::Executor exec,
652 const NeoN::UnstructuredMesh& nfMesh,
653 const FoamFieldType& in
654)
655{
656 using ContainerType = typename TypeMap<FoamFieldType>::container_type;
657 using MappedType = typename TypeMap<FoamFieldType>::mapped_type;
658
659 if constexpr (NeoFOAM::detail::isVolumeField<ContainerType>)
660 {
661 using FoamValueType = typename FoamFieldType::value_type;
662 ContainerType out(exec, in.name(), nfMesh, readVolBoundaryConditions(nfMesh, in));
663 out.internalVector() = fromFoamField(exec, in.primitiveField());
664 std::size_t nBnd = 0;
665 forAll(in.boundaryField(), patchi)
666 {
667 nBnd += in.boundaryField()[patchi].size();
668 }
669
670 Foam::Field<FoamValueType> bval(nBnd);
671
672 Foam::label bi = 0;
673 forAll(in.boundaryField(), patchi)
674 {
675 const auto& pin = in.boundaryField()[patchi];
676 forAll(pin, i)
677 {
678 // IMPORTANT:
679 // keep OpenFOAM type here (scalar or vector)
680 bval[bi++] = pin[i];
681 }
682 }
683
684 NF_ASSERT_EQUAL(static_cast<std::size_t>(bi), nBnd);
685 out.boundaryData().value() = fromFoamField(exec, bval);
686 out.correctBoundaryConditions();
687 return out;
688 }
689 else if constexpr (NeoFOAM::detail::isSurfaceField<ContainerType>)
690 {
691 // Element type of the GeometricField (T in GeometricField<T,...>) — NOT cmptType,
692 // which decomposes vectors into scalars and would break the surface-vector path.
693 using FoamComponentType = typename FoamFieldType::value_type;
694
695 ContainerType out(exec, in.name(), nfMesh, readSurfaceBoundaryConditions(nfMesh, in));
696
697 const std::size_t nInt = nfMesh.nInternalFaces();
698 const std::size_t nBnd = nfMesh.boundaryMesh().offset().back();
699
700 NF_DINFO("Internal: " + std::to_string(nInt) + ", Boundary: " + std::to_string(nBnd));
701
702 Foam::Field<FoamComponentType> internalData(nInt);
703 Foam::Field<FoamComponentType> bval(nBnd);
704
705 // Internal faces only: [0, nInt)
706 forAll(in, facei)
707 {
708 if (static_cast<std::size_t>(facei) < nInt)
709 {
710 internalData[facei] = in[facei];
711 }
712 }
713
714 // Boundary faces in patch order: [0, nBnd)
715 Foam::label bi = 0;
716 // Pass 1 — non-processor patches.
717 forAll(in.boundaryField(), patchi)
718 {
719 const auto& pin = in.boundaryField()[patchi];
720 if (pin.patch().type() == "processor")
721 {
722 continue;
723 }
724 forAll(pin, facei)
725 {
726 bval[bi] = pin[facei];
727 ++bi;
728 }
729 }
730 // Pass 2 — processor patches (proc tail of the boundary range).
731 forAll(in.boundaryField(), patchi)
732 {
733 const auto& pin = in.boundaryField()[patchi];
734 if (pin.patch().type() != "processor")
735 {
736 continue;
737 }
738 forAll(pin, facei)
739 {
740 bval[bi] = pin[facei];
741 ++bi;
742 }
743 }
744
745 NF_ASSERT_EQUAL(static_cast<std::size_t>(bi), nBnd);
746
747 out.internalVector() = fromFoamField(exec, internalData);
748 out.boundaryData().value() = fromFoamField(exec, bval);
749 return out;
750 }
751 else
752 {
753 NF_ASSERT(
754 (!std::is_same_v<ContainerType, ContainerType>),
755 "TypeMap<FoamFieldType>::container_type must be VolumeField<ValueType> or "
756 "SurfaceField<ValueType>."
757 );
758 }
759}
