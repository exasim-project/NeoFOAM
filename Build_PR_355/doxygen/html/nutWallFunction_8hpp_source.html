19static constexpr label MAX_ITER = 10;
20// OpenFOAM nutUSpaldingWallFunction defaults (maxIter=10, tolerance=0.01). The
21// Newton loop must stop at the same err threshold as upstream, otherwise the
22// wall nut converges to a slightly different uTau and the momentum solve drifts.
23static constexpr scalar TOLERANCE = 0.01;
24static constexpr scalar KAPPA = 0.41;
25static constexpr scalar E_COEFF = 9.8;
26
27// tolerance mirrors OpenFOAM's dict-configurable tolerance_ (default 0.01): the
28// Newton loop stops on the relative uTau update, not on full convergence.
29KOKKOS_INLINE_FUNCTION
30scalar computeUTau(
31 const scalar magGradU,
32 const scalar magUp,
33 const scalar y,
34 const scalar nuw,
35 const scalar nutw,
36 scalar& err,
37 const int maxIter,
38 const scalar tolerance
39)
40{
41 err = 0.0;
42
43 scalar ut = Kokkos::sqrt((nutw + nuw) * magGradU);
44 if (ut <= ROOTVSMALL)
45 {
46 return 0.0;
47 }
48
49 int iter = 0;
50 do
51 {
52 const scalar kUu = Kokkos::min(KAPPA * magUp / ut, scalar(50));
53 const scalar fkUu = Kokkos::exp(kUu) - 1.0 - kUu * (1.0 + 0.5 * kUu);
54
55 const scalar f =
56 -ut * y / nuw + magUp / ut + (1.0 / E_COEFF) * (fkUu - (1.0 / 6.0) * kUu * kUu * kUu);
57
58 const scalar df = y / nuw + magUp / (ut * ut) + (1.0 / E_COEFF) * kUu * fkUu / ut;
59
60 const scalar uTauNew = ut + f / df;
61 err = NeoN::mag((ut - uTauNew) / ut);
62 ut = uTauNew;
63 }
64 while (ut > ROOTVSMALL && err > tolerance && ++iter < maxIter);
65
66 return ut > 0.0 ? ut : 0.0;
67}

69inline void setNutUSpaldingWallFunction(
70 Field<scalar>& domainVector,
71 const fvcc::VolumeField<Vec3>& u,
72 const fvcc::VolumeField<scalar>& nu,
73 const fvcc::VolumeField<scalar>& nearWallDist,
74 const UnstructuredMesh& mesh,
75 std::pair<localIdx, localIdx> range
76)
77{
78 const auto uInternal = u.internalVector().view();
79 const auto uBoundary = u.boundaryData().value().view();
80 const auto nuBoundary = nu.boundaryData().value().view();
81
82 auto [refGradient, value, valueFraction, refValue, faceCells, deltaCoeffs, delta] = views(
83 domainVector.boundaryData().refGrad(),
84 domainVector.boundaryData().value(),
85 domainVector.boundaryData().valueFraction(),
86 domainVector.boundaryData().refValue(),
87 mesh.boundaryMesh().faceOwners(),
88 mesh.boundaryMesh().deltaCoeffs(),
89 nearWallDist.boundaryData().value()
90 );
91 NeoN::parallelFor(
92 domainVector.exec(),
93 range,
94 NEON_LAMBDA(const localIdx i) {
95 const localIdx owner = faceCells[i];
96
97 const Vec3 uInt = uInternal[owner];
98 const Vec3 uWall = uBoundary[i];
99 const Vec3 diff = uInt - uWall;
100
101 const scalar magUp = NeoN::mag(diff);
102 const scalar magGradU = NeoN::mag(diff * deltaCoeffs[i]);
103 const scalar y = delta[i];
104 const scalar nuw = nuBoundary[i];
105
106 const scalar currentNut = value[i];
107
108 scalar err = 0.0;
109 const scalar uTau =
110 computeUTau(magGradU, magUp, y, nuw, currentNut, err, MAX_ITER, TOLERANCE);
111
112 // OF nutUSpaldingWallFunctionFvPatchScalarField::calcNut: at the default
113 // tolerance (0.01) the restart-preservation branch (kept only for a
114 // user-overridden tolerance) is skipped, so nutw is taken straight from
115 // the freshly-solved uTau: max(0, uTau^2/magGradU - nuw).
116 const scalar nutCandidate = (uTau * uTau) / (magGradU + ROOTVSMALL) - nuw;
117 const scalar nutw = nutCandidate > 0.0 ? nutCandidate : 0.0;
118
119 refValue[i] = nutw;
120 value[i] = nutw;
121 valueFraction[i] = 1.0;
122 refGradient[i] = 0.0;
123 },
124 "setNutUSpaldingWallFunction"
125 );
126}
