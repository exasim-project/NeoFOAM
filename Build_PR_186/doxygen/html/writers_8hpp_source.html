20/*@brief copy from neon src vector on device to dest OF field*/
21template<class SrcValueType, class DestField>
22void copyImpl(const NeoN::Vector<SrcValueType>& src, DestField& dest)
23{
24 NF_ASSERT_EQUAL(dest.size(), src.size());
25 auto srcHost = src.copyToHost();
26 auto srcView = srcHost.view();
27 for (int i = 0; i < dest.size(); i++)
28 {
29 dest[i] = convert(srcView[i]);
30 }
31}
