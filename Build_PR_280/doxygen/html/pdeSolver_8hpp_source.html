78 struct SetReference : public NeoN::dsl::PostAssemblyBase<ValueType, IndexType>

88
89 virtual void operator()(NeoN::la::LinearSystem<
90 FunctorValueType,
91 NeoN::la::CSRMatrix<FunctorValueType, IndexType>>& ls)
92 {
93 // `pRefCell_` is a local cell index on the rank that owns the
94 // reference cell. All other ranks receive pRefCell = -1 from
95 // OpenFOAM's setRefCell() and the `pRefCell >= 0` gate at the
96 // neoIcoFoam call site ensures setReference() is only called on
97 // the owning rank. No MPI rank guard is needed here.
98
99 const auto rowOffs = ls.matrix().sparsity()->rowOffs().view();
100 const auto diagOffset = ls.faceToMatrixAddress()->diagOffset().view();
101 auto rhs = ls.rhs().view();
102 auto values = ls.matrix().values().view();
103 // make an explicit copy to avoid capture this warning in kokkos lambda
104 auto pRefValue = pRefValue_;
105
106 NeoN::parallelFor(
107 ls.exec(),
108 {pRefCell_, pRefCell_ + 1},
109 NEON_LAMBDA(const std::size_t refCelli) {
110 auto diagIdx = rowOffs[refCelli] + diagOffset[refCelli];
111 auto diagValue = values[diagIdx];
112 rhs[refCelli] += diagValue * pRefValue;
113 values[diagIdx] += diagValue;
114 }
115 );
116 }
117 };

155 NeoN::la::LinearSystem<ValueType> assemble(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
156 {
157 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
158 rhsExpr.read(runTime_.fvSchemesDict);
159 auto ls = NeoN::la::LinearSystem<ValueType>(assemble());
160
161 auto expTmp = rhsExpr.explicitOperation(psi_.mesh().nCells());
162
163 auto [vol, expSource, rhsV] = NeoN::views(psi_.mesh().cellVolumes(), expTmp, ls.rhs());
164 NeoN::parallelFor(
165 psi_.exec(),
166 {0, rhsV.size()},
167 NEON_LAMBDA(const NeoN::localIdx i) { rhsV[i] -= expSource[i] * vol[i]; }
168 );
169
170 return ls;
171 }

181 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
182 {
183 // assemble wo rhs first
184 auto ls = assemble(std::move(rhs));
185
186 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
187 auto fvSolution = solverDict.subDict(psi_.name);
188 auto solver = NeoN::la::Solver(psi_.exec(), fvSolution);
189 // Do some sanity checks before trying to solve
190 // NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
191 auto stats = solver.solve(ls, psi_.internalVector());
192
193 for (auto& stat : stats.entries)
194 {
195 NeoN::Logging::info(
196 "Solving for {} Initial residual: {} Final residual: {} No Iterations: {}",
197 psi_.name,
198 stat.initResNorm,
199 stat.finalResNorm,
200 stat.numIter
201 );
202 }
203 return stats;
204 }

286template<typename ValueType, typename IndexType = NeoN::localIdx>
287NeoN::finiteVolume::cellCentred::VolumeField<ValueType> applyOperator(
288 const la::LinearSystem<ValueType, NeoN::la::CSRMatrix<ValueType, IndexType>>& ls,
289 const NeoN::finiteVolume::cellCentred::VolumeField<ValueType>& psi
290)
291{
292 NeoN::finiteVolume::cellCentred::VolumeField<ValueType> res(
293 psi.exec(),
294 "ls_" + psi.name,
295 psi.mesh(),
296 psi.internalVector(),
297 psi.boundaryData(),
298 psi.boundaryConditions()
299 );
300 NeoN::la::computeResidual(ls.matrix(), ls.rhs(), psi.internalVector(), res.internalVector());
301 return res;
302}
