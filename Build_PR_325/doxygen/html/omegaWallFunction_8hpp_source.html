29// Applies the omegaWallFunction kernel face-by-face.
30//
31// Matches Foam::omegaWallFunctionFvPatchScalarField::calculate() for the
32// default blender (BINOMIAL, n=2) and updateCoeffs()'s write-back step
33// (omega.C:485-527): each wall face contributes a blended omega to its
34// owner cell and the BC face value is the same blended omega.
35//
36// Limitations vs upstream:
37// - Single-patch corner only: we assume each wall cell sees at most one
38// wall-function face, i.e. cornerWeights == 1. Multi-patch corner
39// averaging (omega.C:71-126) is not modelled — for typical meshes where
40// interior cells are bounded by ≤1 wall face this is exact; meshes with
41// re-entrant corners shared by two wall patches will diverge.
42// - G (production) accumulation is not applied here. Upstream's
43// calculate() adds (νₜ+ν)·|∇U_w|·C_µ^0.25·√k/(κ·y) into the turbulence
44// model's G field; NeoFOAM's kOmegaSST computes G internally from
45// nut·GbyNu0 and does not yet route it through boundary BCs. The
46// momentum/k near-wall production therefore won't match upstream
47// exactly even when omega values agree.
48inline void setOmegaWallFunction(
49 Field<scalar>& omega,
50 const fvcc::VolumeField<scalar>& k,
51 const fvcc::VolumeField<scalar>& nu,
52 const fvcc::VolumeField<scalar>& nearWallDist,
53 const UnstructuredMesh& mesh,
54 std::pair<localIdx, localIdx> range,
55 scalar beta1,
56 scalar Cmu,
57 scalar kappa
58)
59{
60 const scalar Cmu25 = Kokkos::pow(Cmu, scalar(0.25));
61
62 auto kInternal = k.internalVector().view();
63 const auto nuBoundary = nu.boundaryData().value().view();
64 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
65
66 auto [refGrad, value, valueFraction, refValue, faceOwners] = views(
67 omega.boundaryData().refGrad(),
68 omega.boundaryData().value(),
69 omega.boundaryData().valueFraction(),
70 omega.boundaryData().refValue(),
71 mesh.boundaryMesh().faceOwners()
72 );
73
74 NeoN::parallelFor(
75 omega.exec(),
76 range,
77 NEON_LAMBDA(const localIdx i) {
78 const localIdx owner = faceOwners[i];
79 const scalar y = nearWallBoundary[i];
80 const scalar nuw = nuBoundary[i];
81 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
82
83 // Viscous sublayer: 6ν / (β₁·y²) (omega.C:218-224)
84 const scalar wVis = 6.0 * nuw / (beta1 * y * y);
85
86 // Log layer: √k / (C_µ^0.25·κ·y) (omega.C:226-234)
87 const scalar wLog = Kokkos::sqrt(kw) / (Cmu25 * kappa * y);
88
89 // BINOMIAL blender, n = 2 (upstream default at omega.C:396) —
90 // closed form: ω = √(ωᵥᵢₛ² + ωₗₒg²).
91 const scalar wOmega = Kokkos::sqrt(wVis * wVis + wLog * wLog);
92
93 // Set the wall face value only. Upstream also writes
94 // omega_internal[wall_cell] via manipulateMatrix(setValues) — a
95 // matrix-row pin we have no equivalent for. Without that pin a
96 // direct stomp on omega.internalVector() only persists until the
97 // next omega solve overwrites it; meanwhile correctNutInternal
98 // reads the stomped value and collapses nut.internal at the wall
99 // cell. Leaving the internal value alone keeps the fixedValue
100 // semantics correct via the boundary face.
101 value[i] = wOmega;
102 refValue[i] = wOmega;
103 valueFraction[i] = 1.0;
104 refGrad[i] = 0.0;
105 },
106 "setOmegaWallFunction"
107 );
108}

113// Mirrors Foam::omegaWallFunctionFvPatchScalarField
114// (src/TurbulenceModels/turbulenceModels/derivedFvPatchFields/wallFunctions/
115// omegaWallFunctions/omegaWallFunction/omegaWallFunctionFvPatchScalarField.{H,C}).
116//
117// Upstream is a fixedValueFvPatchField<scalar> whose updateCoeffs() blends a
118// viscous-sublayer and a log-layer estimate of omega and also overwrites the
119// adjacent wall cell's omega in the internal field. We replicate that here
120// inside correctBoundaryCondition(ctx).
121//
122// Context fields required:
123// - "k" VolumeField<scalar> — to evaluate √k at the wall cell
124// - "nu" VolumeField<scalar> — laminar viscosity, boundary face values
125// - "nearWallDist" VolumeField<scalar> — y for the wall cell, boundary face values
126//
127// Mirroring NutUSpaldingWallFunction (nutWallFunction.hpp:142-153), the no-arg
128// correctBoundaryCondition() overload is a no-op: this BC must be driven from
129// kOmegaSST::correct() (or equivalent) with a BoundaryContext that supplies
130// the three fields above.
131class OmegaWallFunction : public VolumeBoundaryFactory<scalar>::template Register<OmegaWallFunction>
132{
133 using Base = VolumeBoundaryFactory<scalar>::template Register<OmegaWallFunction>;
134
135public:
136
137 OmegaWallFunction(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
138 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = true})
139 , mesh_(mesh)
140 , beta1_(
141 dict.contains("beta1") ? dict.get<scalar>("beta1") : detail::OMEGA_WF_DEFAULT_BETA1
142 )
143 , Cmu_(dict.contains("Cmu") ? dict.get<scalar>("Cmu") : detail::OMEGA_WF_DEFAULT_CMU)
144 , kappa_(
145 dict.contains("kappa") ? dict.get<scalar>("kappa") : detail::OMEGA_WF_DEFAULT_KAPPA
146 )
147 {}
148
149 void correctBoundaryCondition(Field<scalar>& /*domainVector*/) final {}
150
151 void
152 correctBoundaryCondition(Field<scalar>& domainVector, const fvcc::BoundaryContext& ctx) final
153 {
154 detail::setOmegaWallFunction(
155 domainVector,
156 ctx.scalarFieldPtr("k"),
157 ctx.scalarFieldPtr("nu"),
158 ctx.scalarFieldPtr("nearWallDist"),
159 mesh_,
160 this->range(),
161 beta1_,
162 Cmu_,
163 kappa_
164 );
165 }
