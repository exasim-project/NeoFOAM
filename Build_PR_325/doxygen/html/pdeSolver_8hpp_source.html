199 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
200 {
201 // Read-path fix: relax the OWNED ls_ IN PLACE so computeRAUandHByA — which reads
202 // ls_ (pressureVelocityCoupling.cpp) — sees the RELAXED augmented diagonal, mirroring
203 // OpenFOAM's UEqn.relax() boosting A() ~1/alpha in place. The system actually SOLVED stays
204 // identical to today's (relaxed-diag + relax-source + (-grad p)) to machine precision:
205 // today adds -grad p to the copy THEN relaxes the copy; here ls_ is relaxed BEFORE the
206 // copy+add. Matrix-URF touches only diag + relax-source (not the explicit -grad p rhs
207 // source), so nfU is unchanged → test_momentum stays green. The ONLY new behavior: ls_ now
208 // carries the relaxed diagonal for the rAU/HbyA read.
209
210 // 1) Assemble the OWNED ls_ (raw momentum predictor, no rhs term yet) and relax it in
211 // place. The lookup+relax is factored into relaxOwnedLs() so the no-predictor path
212 // (assembleAndRelax()) shares the SAME alpha lookup and relaxation and the two
213 // cannot drift. A missing factor -> 1.0 -> applyMatrixRelaxation early-returns (bitwise
214 // no-op for cases without U relax, i.e. neoIcoFoam/neoPisoFoam). Relaxation runs on the
215 // RAW assembled diagonal (relax before setReference), exactly as solveImpl.
216 assemble();
217 relaxOwnedLs();
218
219 // 2) (perf) Apply -grad p to the OWNED ls_'s rhs IN PLACE for
220 // the solve instead of DEEP-COPYING the whole momentum LinearSystem. The matrix is the
221 // heavy part (~nnz Vec3 entries, ~1 GB at 6.3M cells) and is NOT modified by adding the
222 // pressure gradient — assembleExplicitSource touches rhs only, and Solver::solve takes the
223 // system as const — so solving ls_ here is byte-identical to solving the old copy. Only the
224 // rhs differs (O(nCells)), so we snapshot the rhs-free relaxed-momentum ("H") rhs, apply
225 // the source, solve ls_, then RESTORE the snapshot. After restore, ls_ holds exactly what
226 // it did before (relaxed diagonal + bare momentum rhs, no -grad p) so the subsequent
227 // computeRAUandHByA reads the H-system (the rAU/HbyA read invariant). ls_ is relaxed once
228 // (step 1); not re-relaxed.
229 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
230 rhsExpr.read(runTime_.fvSchemesDict);
231 auto savedRhs = NeoN::Vector<ValueType>(ls_.rhs());
232 rhsExpr.assembleExplicitSource(ls_, psi_.mesh());
233
234 // 3) select the <field>Final solver subdict on the final outer pass;
235 // isDict-guarded fallback to the base subdict, identical to solveImpl.
236 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
237 const std::string finalKey = psi_.name + "Final";
238 auto fvSolution = (finalIter_ && solverDict.isDict(finalKey))
239 ? solverDict.subDict(finalKey)
240 : solverDict.subDict(psi_.name);
241 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
242 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
243 stripNeoFOAMKeys(fvSolution);
244 auto solver = NeoN::la::Solver(psi_.exec(), fvSolution);
245 // Do some sanity checks before trying to solve
246 // NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
247 auto stats = solver.solve(ls_, psi_.internalVector());
248
249 // Restore the rhs-free momentum rhs so computeRAUandHByA reads the H-system, not H - grad
250 // p.
251 ls_.rhs() = savedRhs;
252
253 reportSolverStats(stats, fvSolution);
254 return stats;
255 }

257 // Public because NVCC forbids extended __host__ __device__ lambdas
258 // (NEON_LAMBDA) inside private or protected member functions.
259 NeoN::la::SolverStats solveImpl(dsl::Expression<ValueType>& expr, LinearSystem& ls)
260 {
261 // Re-read schemes (idempotent with the constructor read)
262 expr.read(runTime_.fvSchemesDict);
263
264 // Assemble without post-assembly functors; we apply SetReference separately below
265 // to ensure correct polymorphic dispatch — storing PostAssemblyBase by value causes
266 // object slicing that silently disables virtual overrides.
267 expr.assemble(runTime_.t, runTime_.dt, ls, psi_.mesh());
268
269 // Equation (matrix) under-relaxation, applied directly here mirroring the
270 // SetReference direct call, NOT via a PostAssemblyBase ps vector. The
271 // factor is parsed from fvSolution in NeoFOAM; NeoN only sees a scalar.
272 // alpha is a plain scalar, so this runs for both scalar and Vec3 fields and lives
273 // OUTSIDE the SetReference `if constexpr` guard. A missing entry -> 1.0 -> the
274 // kernel early-returns (bitwise no-op). psi_ supplies psi_prev (internalVector)
275 // AND the mesh (faceOwners() for the boundary-diagonal owner).
276 //
277 // Relaxation MUST run on the raw assembled diagonal, BEFORE SetReference
278 // (OpenFOAM order: fvMatrix::relax() precedes setReference). SetReference doubles the
279 // ref cell's diagonal and adds diag*refVal to rhs; relaxing afterwards would
280 // reconstruct/clamp the already-doubled diagonal and corrupt the pin if a field ever
281 // carried both a reference cell and a relaxation factor.
282 const auto alpha =
283 lookupEqnRelaxation(runTime_.fvSolutionDict, psi_.name, finalIter_).value_or(1.0);
284 NeoN::dsl::applyMatrixRelaxation(ls, psi_, alpha);
285
286 // Apply reference-cell pinning directly (avoids object-slicing issue)
287 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
288 {
289 if (needReference_)
290 {
291 NeoN::dsl::SetReference<ValueType> refFunct(pRefCell_, pRefValue_);
292 refFunct(ls);
293 }
294 }
295
296 // select the <field>Final solver subdict on the final outer pass (tighter
297 // pimpleFoam final-pass tolerances); isDict-guarded fallback to the base
298 // subdict (a missing/non-dict <field>Final degrades to the base entry instead
299 // of throwing out of subDict).
300 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
301 const std::string finalKey = psi_.name + "Final";
302 auto fieldSolverDict = (finalIter_ && solverDict.isDict(finalKey))
303 ? solverDict.subDict(finalKey)
304 : solverDict.subDict(psi_.name);
305 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
306 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
307 stripNeoFOAMKeys(fieldSolverDict);
308 NeoN::fence(psi_.exec());
309 NF_ASSERT(ls.exec() == psi_.exec(), "Executors are not the same");
310
311 auto solver = NeoN::la::Solver(psi_.exec(), fieldSolverDict);
312 auto stats = solver.solve(ls, psi_.internalVector());
313
314 reportSolverStats(stats, fieldSolverDict);
315
316 return stats;
317 }
318
319 // Relax the OWNED ls_ in place using the production equation-URF lookup
320 // (relaxationFactors.equations.<field>[Final]). Shared verbatim by solve(rhs)
321 // and assembleAndRelax() so the predictor and no-predictor paths cannot diverge.
322 // alpha defaults to 1.0 (missing factor) -> applyMatrixRelaxation early-returns: a bitwise
323 // no-op preserving neoIcoFoam/neoPisoFoam semantics. Mirrors UEqn.relax() and MUST run on
324 // the raw assembled diagonal BEFORE any SetReference. Only CALLS the free
325 // function applyMatrixRelaxation (which owns the NEON_LAMBDA) — no extended lambda is defined
326 // in this member body, so the NVCC private-member-lambda restriction does not apply.
327 void relaxOwnedLs()
328 {
329 const auto alpha =
330 lookupEqnRelaxation(runTime_.fvSolutionDict, psi_.name, finalIter_).value_or(1.0);
331 NeoN::dsl::applyMatrixRelaxation(ls_, psi_, alpha);
332 }

336 // Resolve the equation under-relaxation factor (alpha) for the given field
337 // from fvSolution.relaxationFactors.equations.<fieldName>. Returns 1.0 if
338 // the block, the equations subdict, or the field entry is absent — which
339 // applyMatrixRelaxation short-circuits as a bitwise no-op. Tolerates the
340 // entry being typed as either NeoN::scalar (e.g. `U 0.9;`) or int (`U 1;`).
341 static NeoN::scalar
342 readEqnRelaxationFactor(const NeoN::Dictionary& fvSolutionDict, const std::string& fieldName)
343 {
344 if (!fvSolutionDict.contains("relaxationFactors")) return NeoN::scalar(1);
345 const auto& rf = fvSolutionDict.subDict("relaxationFactors");
346 if (!rf.contains("equations")) return NeoN::scalar(1);
347 const auto& eqs = rf.subDict("equations");
348 if (!eqs.contains(fieldName)) return NeoN::scalar(1);
349 if (eqs.isType<int>(fieldName))
350 {
351 return NeoN::scalar(eqs.get<int>(fieldName));
352 }
353 return eqs.get<NeoN::scalar>(fieldName);
354 }

370 // Per-component name (Ux/Uy/Uz) when a vector field is solved as separate
371 // component systems; the plain field name otherwise (scalar or coupled solve).
372 static std::string componentName(const std::string& base, std::size_t i, std::size_t n)
373 {
374 if (n <= 1) return base;
375 constexpr const char* suffix[3] = {"x", "y", "z"};
376 return i < 3 ? base + suffix[i] : base + "[" + std::to_string(i) + "]";

379 // OpenFOAM-style per-solve residual report: "<precond><solver>: Solving for
380 // <field>, Initial residual = ..., Final residual = ..., No Iterations N".
381 // The "<precond><solver>" label (e.g. DICPCG) is read from the solver dict's
382 // reportName meta key stashed by mapFvSolution -> no recomputation per solve.
383 // Logging is rank-aware and skips formatting on muted ranks (NeoN shouldLog).
384 void reportSolverStats(
385 const NeoN::la::SolverStats& stats,
386 const NeoN::Dictionary& fieldSolverDict
387 ) const
388 {
389 const std::string label = fieldSolverDict.contains("reportName")
390 ? fieldSolverDict.get<std::string>("reportName")
391 : std::string("Ginkgo");
392 const std::size_t n = stats.entries.size();
393 for (std::size_t i = 0; i < n; ++i)
394 {
395 const auto& stat = stats.entries[i];
396 NeoN::Logging::info(
397 "{}: Solving for {}, Initial residual = {}, Final residual = {}, No Iterations {}",
398 label,
399 componentName(psi_.name, i, n),
400 stat.initResNorm,
401 stat.finalResNorm,
402 stat.numIter
403 );
404 }
405 }

420template<typename ValueType, typename IndexType = NeoN::localIdx>
421NeoN::finiteVolume::cellCentred::VolumeField<ValueType> applyOperator(
422 const la::LinearSystem<ValueType>& ls,
423 const NeoN::finiteVolume::cellCentred::VolumeField<ValueType>& psi
424)
425{
426 NeoN::finiteVolume::cellCentred::VolumeField<ValueType> res(
427 psi.exec(),
428 "ls_" + psi.name,
429 psi.mesh(),
430 psi.internalVector(),
431 psi.boundaryData(),
432 psi.boundaryConditions()
433 );
434 NeoN::la::computeResidual(ls.matrix(), ls.rhs(), psi.internalVector(), res.internalVector());
435 return res;
436}
