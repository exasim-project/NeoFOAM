278 // NO-OP WITHOUT A POOL: resize(0)+regrow without a pool is a raw cudaMalloc/cudaFree pair
279 // per step — costlier than keeping the field resident.
280 template<class... Vs>
281 void freeVecs(Vs&... vs)
282 {
283 if (!devicePoolActive()) return;
284 NeoN::fence(exec_);
285 auto drop = [](auto& v)
286 {
287 if (v.size() != 0) v.resize(0);
288 };
289 (drop(vs), ...);
290 }
