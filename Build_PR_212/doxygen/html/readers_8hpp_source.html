32template<typename FoamType>
33auto readVolBoundaryConditions(const NeoN::UnstructuredMesh& nfMesh, const FoamType& ofVolField)
34{
35 using type_container_t = typename TypeMap<FoamType>::container_type;
36 using type_primitive_t = typename TypeMap<FoamType>::mapped_type;
37
38 // get boundary as dictionary
39 Foam::OStringStream os;
40 ofVolField.boundaryField().writeEntries(os);
41 Foam::IStringStream is(os.str());
42 Foam::dictionary bDict(is);
43
44 std::map<std::string, std::function<void(NeoN::Dictionary&)>> patchInserter {
45 {"fixedGradient",
46 [](auto& dict)
47 {
48 dict.insert("type", std::string("fixedGradient"));
49 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("value");
50 type_primitive_t fixedGradient = tokenList.get<type_primitive_t>(1);
51 dict.insert("fixedGradient", fixedGradient);
52 }},
53 {"zeroGradient",
54 [&](auto& dict)
55 {
56 dict.insert("type", std::string("fixedGradient"));
57 dict.insert("fixedGradient", type_primitive_t {});
58 }},
59 {"fixedValue",
60 [](auto& dict)
61 {
62 dict.insert("type", std::string("fixedValue"));
63 NeoN::TokenList tokenList = dict.template get<NeoN::TokenList>("value");
64 type_primitive_t fixedValue {};
65 // test if things can be read as scalar first, if it doesn't work
66 // read as int and convert to scalar
67 if constexpr (std::is_same<type_primitive_t, NeoN::Vec3>::value)
68 {
69 NeoN::Vec3 tmpFixedValue {};
70 auto tokens = tokenList.tokens();
71 NeoN::scalar* ret = std::any_cast<NeoN::scalar>(&tokens[1]);
72 if (ret)
73 {
74 tmpFixedValue[0] = tokenList.get<NeoN::scalar>(1);
75 tmpFixedValue[1] = tokenList.get<NeoN::scalar>(2);
76 tmpFixedValue[2] = tokenList.get<NeoN::scalar>(3);
77 }
78 else
79 {
80 tmpFixedValue[0] = NeoN::scalar(tokenList.get<int>(1));
81 tmpFixedValue[1] = NeoN::scalar(tokenList.get<int>(2));
82 tmpFixedValue[2] = NeoN::scalar(tokenList.get<int>(3));
83 }
84 dict.insert("fixedValue", tmpFixedValue);
85 }
86 else
87 {
88 auto& token = tokenList.tokens()[1];
89 NeoN::scalar* ret = std::any_cast<NeoN::scalar>(&token);
90 fixedValue = ret ? NeoN::scalar(*ret) : NeoN::scalar(std::any_cast<int>(token));
91 dict.insert("fixedValue", fixedValue);
92 }
93 }},
94 {"noSlip", // TODO specialize for vector
95 [](auto& dict)
96 {
97 dict.insert("type", std::string("fixedValue"));
98 dict.insert("fixedValue", type_primitive_t {});
99 }},
100 {"calculated", [](auto& dict) { dict.insert("type", std::string("calculated")); }},
101 {"extrapolatedCalculated",
102 [](auto& dict) { dict.insert("type", std::string("calculated")); }},
103 {"empty", [](auto& dict) { dict.insert("type", std::string("empty")); }},
104 {"symmetryPlane", [](auto& dict) { dict.insert("type", std::string("symmetry")); }},
105 {"symmetry", [](auto& dict) { dict.insert("type", std::string("symmetry")); }}
106 };
107
108 int patchi = 0;
109 std::vector<fvcc::VolumeBoundary<type_primitive_t>> bcs;
110 for (const auto& bName : bDict.toc())
111 {
112 Foam::dictionary patchDict = bDict.subDict(bName);
113 NeoN::Dictionary neoPatchDict = convert(patchDict);
114 patchInserter[patchDict.get<Foam::word>("type")](neoPatchDict);
115 bcs.emplace_back(nfMesh, neoPatchDict, patchi);
116 patchi++;
117 }
118 return bcs;
119}

121template<typename FoamType>
122auto readSurfaceBoundaryConditions(
123 const NeoN::UnstructuredMesh& uMesh,
124 const FoamType& surfaceField
125)
126{
127 using type_container_t = typename TypeMap<FoamType>::container_type;
128 using type_primitive_t = typename TypeMap<FoamType>::mapped_type;
129
130 std::vector<fvcc::SurfaceBoundary<type_primitive_t>> bcs;
131
132 // get boundary as dictionary
133 Foam::OStringStream os;
134 surfaceField.boundaryField().writeEntries(os);
135 Foam::IStringStream is(os.str());
136 Foam::dictionary bDict(is);
137 int patchi = 0;
138
139 std::map<std::string, std::function<void(NeoN::Dictionary&)>> patchInserter {
140 {"fixedGradient", [](auto& dict) { dict.insert("type", std::string("fixedGradient")); }},
141 {"zeroGradient",
142 [&](auto& dict)
143 {
144 dict.insert("type", std::string("fixedGradient"));
145 dict.insert("fixedGradient", type_primitive_t {});
146 }},
147 {"fixedValue",
148 [](auto& dict)
149 {
150 dict.insert("type", std::string("fixedValue"));
151 dict.insert("fixedValue", type_primitive_t {});
152 }},
153 {"calculated", [](auto& dict) { dict.insert("type", std::string("calculated")); }},
154 {"empty", [](auto& dict) { dict.insert("type", std::string("empty")); }},
155 {"symmetryPlane", [](auto& dict) { dict.insert("type", std::string("symmetry")); }},
156 {"symmetry", [](auto& dict) { dict.insert("type", std::string("symmetry")); }}
157 };
158
159 for (const auto& bName : bDict.toc())
160 {
161 Foam::dictionary patchDict = bDict.subDict(bName);
162 NeoN::Dictionary neoPatchDict;
163 patchInserter[patchDict.get<Foam::word>("type")](neoPatchDict);
164 bcs.push_back(fvcc::SurfaceBoundary<type_primitive_t>(uMesh, neoPatchDict, patchi));
165 patchi++;
166 }
167 return bcs;
168}

170template<class FoamFieldType>
171auto constructFrom(
172 const NeoN::Executor exec,
173 const NeoN::UnstructuredMesh& nfMesh,
174 const FoamFieldType& in
175)
176{
177 using ContainerType = typename TypeMap<FoamFieldType>::container_type;
178 using MappedType = typename TypeMap<FoamFieldType>::mapped_type;
179
180 if constexpr (NeoFOAM::detail::isVolumeField<ContainerType>)
181 {
182 ContainerType out(exec, in.name(), nfMesh, readVolBoundaryConditions(nfMesh, in));
183 out.internalVector() = fromFoamField(exec, in.primitiveField());
184 out.correctBoundaryConditions();
185 return out;
186 }
187 else if constexpr (NeoFOAM::detail::isSurfaceField<ContainerType>)
188 {
189 using FoamComponentType = typename FoamFieldType::cmptType;
190
191 ContainerType out(exec, in.name(), nfMesh, readSurfaceBoundaryConditions(nfMesh, in));
192
193 const std::size_t nInt = nfMesh.nInternalFaces();
194 const std::size_t nBnd = nfMesh.boundaryMesh().offset().back();
195 const std::size_t nFaces = nInt + nBnd;
196
197 NF_DINFO(
198 "Internal: " + std::to_string(nInt) + ", Boundary: " + std::to_string(nBnd)
199 + ", nFaces: " + std::to_string(nFaces)
200 );
201
202 Foam::Field<FoamComponentType> flat(nFaces);
203 Foam::Field<FoamComponentType> bval(nBnd);
204
205 // Internal faces first: [0, nInt)
206 forAll(in, facei)
207 {
208 if (static_cast<std::size_t>(facei) < nInt)
209 {
210 flat[facei] = convert(in[facei]);
211 }
212 }
213
214 // Boundary faces appended in patch order: [nInt, nFaces)
215 Foam::label idx = static_cast<Foam::label>(nInt);
216 Foam::label bi = 0;
217 forAll(in.boundaryField(), patchi)
218 {
219 const auto& pin = in.boundaryField()[patchi];
220 forAll(pin, facei)
221 {
222 flat[idx] = convert(pin[facei]);
223 bval[bi] = convert(pin[facei]);
224 ++idx;
225 ++bi;
226 }
227 }
228
229 // (Optional) asserts in debug:
230 NF_ASSERT_EQUAL(static_cast<std::size_t>(idx), nFaces);
231 NF_ASSERT_EQUAL(static_cast<std::size_t>(bi), nBnd);
232
233 out.internalVector() = fromFoamField(exec, flat);
234 out.boundaryData().value() = fromFoamField(exec, bval);
235 out.correctBoundaryConditions();
236 return out;
237 }
238 else
239 {
240 NF_ASSERT(
241 (!std::is_same_v<ContainerType, ContainerType>),
242 "TypeMap<FoamFieldType>::container_type must be VolumeField<ValueType> or "
243 "SurfaceField<ValueType>."
244 );
245 }

254template<typename FieldType>
255class CreateFromFoamField
256{
257public:
258
259 const NeoN::Executor exec;
260 const NeoN::UnstructuredMesh& nfMesh;
261 const FieldType& foamField;
262 std::string name = "";
263 std::int64_t iterationIndex = 0;
264 std::int64_t subCycleIndex = -1;
265
266 fvcc::VectorDocument operator()(NeoN::Database& db)
267 {
268 using type_container_t = typename TypeMap<FieldType>::container_type;
269 type_container_t convertedField = constructFrom(exec, nfMesh, foamField);
270 if (name != "")
271 {
272 convertedField.name = name;
273 }
274 const Foam::fvMesh& mesh = foamField.mesh();
275 const Foam::Time& runTime = mesh.time();
276 std::int64_t timeIndex = runTime.timeIndex();
277
278 NeoN::Field<typename type_container_t::VectorValueType> field(
279 convertedField.exec(),
280 convertedField.internalVector(),
281 convertedField.boundaryData()
282 );
283
284 type_container_t registeredField(
285 convertedField.exec(),
286 convertedField.name,
287 convertedField.mesh(),
288 field,
289 convertedField.boundaryConditions(),
290 db,
291 "",
292 ""
293 );
294
295 return NeoN::Document(
296 {{"name", convertedField.name},
297 {"timeIndex", timeIndex},
298 {"iterationIndex", iterationIndex},
299 {"subCycleIndex", subCycleIndex},
300 {"field", registeredField}},
301 fvcc::validateVectorDoc
302 );
303 }
304};
