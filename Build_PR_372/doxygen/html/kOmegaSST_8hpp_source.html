177 // NO-OP WITHOUT A POOL: resize(0)+regrow without a pool is a raw cudaMalloc/cudaFree pair
178 // per step — costlier than keeping the field resident.
179 template<class... Vs>
180 void freeVecs(Vs&... vs)
181 {
182 if (!devicePoolActive()) return;
183 NeoN::fence(exec_);
184 auto drop = [](auto& v)
185 {
186 if (v.size() != 0) v.resize(0);
187 };
188 (drop(vs), ...);
189 }
