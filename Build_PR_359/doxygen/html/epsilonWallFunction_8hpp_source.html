37inline void setEpsilonWallFunction(
38 Field<scalar>& epsilon,
39 const fvcc::VolumeField<scalar>& k,
40 const fvcc::VolumeField<scalar>& nu,
41 const fvcc::VolumeField<scalar>& nearWallDist,
42 const UnstructuredMesh& mesh,
43 std::pair<localIdx, localIdx> range,
44 scalar Cmu,
45 scalar kappa
46)
47{
48 const scalar Cmu75 = Kokkos::pow(Cmu, scalar(0.75));
49
50 auto kInternal = k.internalVector().view();
51 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
52 (void)nu; // molecular viscosity only used by the (disabled) viscous-sublayer branch
53
54 auto [refGrad, value, valueFraction, refValue, faceOwners] = views(
55 epsilon.boundaryData().refGrad(),
56 epsilon.boundaryData().value(),
57 epsilon.boundaryData().valueFraction(),
58 epsilon.boundaryData().refValue(),
59 mesh.boundaryMesh().faceOwners()
60 );
61
62 NeoN::parallelFor(
63 epsilon.exec(),
64 range,
65 NEON_LAMBDA(const localIdx i) {
66 const localIdx owner = faceOwners[i];
67 const scalar y = nearWallBoundary[i];
68 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
69
70 // STEPWISE blender (OpenFOAM v2406 default) with lowReCorrection off:
71 // epsilon0 = epsilonLog = Cmu^0.75 k^1.5 / (kappa y). (The viscous
72 // sublayer branch is only taken when lowReCorrection && yPlus < yPlusLam.)
73 const scalar eLog = Cmu75 * Kokkos::pow(kw, scalar(1.5)) / (kappa * y);
74 const scalar eOmega = eLog;
75
76 value[i] = eOmega;
77 refValue[i] = eOmega;
78 valueFraction[i] = 1.0;
79 refGrad[i] = 0.0;
80 },
81 "setEpsilonWallFunction"
82 );
83}
