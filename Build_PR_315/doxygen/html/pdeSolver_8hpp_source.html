124 void setReference(NeoN::localIdx pRefCell, NeoN::scalar pRefValue)
125 {
126 // Accept the call in serial (MPI not initialised — mpiRank is the
127 // sentinel -1, which would compare unequal to 0 if cast to size_t)
128 // and on rank 0 in parallel. SetReference::operator() further guards
129 // the matrix mutation against non-rank-0 in initialised MPI.
130 const auto& mpiEnv = runTime_.mpiEnvironment;
131 if (!mpiEnv.isInitialized() || mpiEnv.rank() == 0)
132 {
133 needReference_ = true;
134 pRefCell_ = pRefCell;
135 pRefValue_ = pRefValue;
136 }

182 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
183 {
184 // 1. assemble the owned system once (implicit + main explicit sources via NeoN assemble).
185 assemble();
186
187 // 2. build the extra rhs term (e.g. -grad(p)). The main expression's explicit sources are
188 // already folded into ls_ by assemble(), so only the extra term is collected here.
189 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
190 rhsExpr.read(runTime_.fvSchemesDict);
191 auto expSource =
192 rhsExpr.explicitOperation(static_cast<NeoN::localIdx>(psi_.mesh().nCells()));
193
194 // 3. apply the explicit source to ls_.rhs() in place for the solve (rhs -= source * vol).
195 {
196 auto [vol, src, rhsV] = NeoN::views(psi_.mesh().cellVolumes(), expSource, ls_.rhs());
197 NeoN::parallelFor(
198 psi_.exec(),
199 {0, rhsV.size()},
200 NEON_LAMBDA(const NeoN::localIdx i) { rhsV[i] -= src[i] * vol[i]; }
201 );
202 }
203
204 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
205 auto fvSolution = solverDict.subDict(psi_.name);
206 stripNeoFOAMKeys(fvSolution);
207 auto solver = NeoN::la::Solver(psi_.exec(), fvSolution);
208 auto stats = solver.solve(ls_, psi_.internalVector());
209 reportSolverStats(stats, fvSolution);
210
211 // 4. restore ls_.rhs() to the rhs-free state so the following rAU/HbyA read the H-system.
212 {
213 auto [vol, src, rhsV] = NeoN::views(psi_.mesh().cellVolumes(), expSource, ls_.rhs());
214 NeoN::parallelFor(
215 psi_.exec(),
216 {0, rhsV.size()},
217 NEON_LAMBDA(const NeoN::localIdx i) { rhsV[i] += src[i] * vol[i]; }
218 );
219 }
220
221 return stats;
222 }
223
224 // Assemble for a solve honouring the matrix-reuse setting: a full matrix+rhs assemble on the
225 // first solve (or when invalidated via markMatrixDirty, or while a setReference pin is active),
226 // otherwise an rhs-only refresh that reuses the cached matrix. The rhs-only path exists for the
227 // same-type (scalar-matrix == field-type) form only; the segregated vector solve, and any
228 // PDESolver with reuse disabled, always re-assembles in full — i.e. behaviour is unchanged
229 // unless enableMatrixReuse() was called.
230 void assembleForSolve(dsl::Expression<ValueType>& expr, LinearSystem& ls)
231 {
232 if constexpr (std::is_same_v<MatrixValueType, ValueType>)
233 {
234 if (matrixReuse_ && matrixAssembled_ && !needReference_)
235 {
236 ls.resetRhs();
237 expr.assembleRhs(ls, psi_.mesh());
238 return;
239 }
240 }
241 ls.reset();
242 expr.assemble(runTime_.t, runTime_.dt, ls, psi_.mesh());
243 matrixAssembled_ = true;
244 }

246 // Public because NVCC forbids extended __host__ __device__ lambdas
247 // (NEON_LAMBDA) inside private or protected member functions.
248 NeoN::la::SolverStats solveImpl(dsl::Expression<ValueType>& expr, LinearSystem& ls)
249 {
250 // Re-read schemes (idempotent with the constructor read)
251 expr.read(runTime_.fvSchemesDict);
252
253 // Assemble without post-assembly functors; we apply SetReference separately below
254 // to ensure correct polymorphic dispatch — storing PostAssemblyBase by value causes
255 // object slicing that silently disables virtual overrides. assembleForSolve honours the
256 // matrix-reuse setting (full matrix+rhs assemble vs rhs-only refresh).
257 assembleForSolve(expr, ls);
258
259 // Apply reference-cell pinning directly (avoids object-slicing issue)
260 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
261 {
262 if (needReference_)
263 {
264 NeoN::dsl::SetReference<ValueType> refFunct(pRefCell_, pRefValue_);
265 refFunct(ls);
266 }
267 }
268
269 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
270 auto fieldSolverDict = solverDict.subDict(psi_.name);
271 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
272 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
273 stripNeoFOAMKeys(fieldSolverDict);
274 NeoN::fence(psi_.exec());
275 NF_ASSERT(ls.exec() == psi_.exec(), "Executors are not the same");
276
277 auto solver = NeoN::la::Solver(psi_.exec(), fieldSolverDict);
278 auto stats = solver.solve(ls, psi_.internalVector());
279
280 reportSolverStats(stats, fieldSolverDict);
281
282 return stats;
283 }

287 // Remove NeoFOAM-specific control keys from a per-field solver dict before it
288 // is passed to NeoN/Ginkgo. Ginkgo's config parser is strict and aborts on any
289 // unrecognised key, so these must be popped here rather than left in place.
290 static void stripNeoFOAMKeys(NeoN::Dictionary& dict)
291 {
292 for (const auto& key : {"assemblyStrategy", "optimize"})
293 {
294 if (dict.contains(key))
295 {
296 dict.remove(key);
297 }
298 }
299 }

301 // Per-component name (Ux/Uy/Uz) when a vector field is solved as separate
302 // component systems; the plain field name otherwise (scalar or coupled solve).
303 static std::string componentName(const std::string& base, std::size_t i, std::size_t n)
304 {
305 if (n <= 1) return base;
306 constexpr const char* suffix[3] = {"x", "y", "z"};
307 return i < 3 ? base + suffix[i] : base + "[" + std::to_string(i) + "]";
308 }

310 // OpenFOAM-style per-solve residual report: "<precond><solver>: Solving for
311 // <field>, Initial residual = ..., Final residual = ..., No Iterations N".
312 // The "<precond><solver>" label (e.g. DICPCG) is read from the solver dict's
313 // reportName meta key stashed by mapFvSolution -> no recomputation per solve.
314 // Logging is rank-aware and skips formatting on muted ranks (NeoN shouldLog).
315 void reportSolverStats(
316 const NeoN::la::SolverStats& stats,
317 const NeoN::Dictionary& fieldSolverDict
318 ) const
319 {
320 const std::string label = fieldSolverDict.contains("reportName")
321 ? fieldSolverDict.get<std::string>("reportName")
322 : std::string("Ginkgo");
323 const std::size_t n = stats.entries.size();
324 for (std::size_t i = 0; i < n; ++i)
325 {
326 const auto& stat = stats.entries[i];
327 NeoN::Logging::info(
328 "{}: Solving for {}, Initial residual = {}, Final residual = {}, No Iterations {}",
329 label,
330 componentName(psi_.name, i, n),
331 stat.initResNorm,
332 stat.finalResNorm,
333 stat.numIter
334 );
335 }
336 }

353template<typename ValueType, typename IndexType = NeoN::localIdx>
354NeoN::finiteVolume::cellCentred::VolumeField<ValueType> applyOperator(
355 const la::LinearSystem<ValueType>& ls,
356 const NeoN::finiteVolume::cellCentred::VolumeField<ValueType>& psi
357)
358{
359 NeoN::finiteVolume::cellCentred::VolumeField<ValueType> res(
360 psi.exec(),
361 "ls_" + psi.name,
362 psi.mesh(),
363 psi.internalVector(),
364 psi.boundaryData(),
365 psi.boundaryConditions()
366 );
367 NeoN::la::computeResidual(ls.matrix(), ls.rhs(), psi.internalVector(), res.internalVector());
368 return res;
369}
