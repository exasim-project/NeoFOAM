26{
27 Foam::FatalError.throwExceptions(true);
28 Foam::FatalIOError.throwExceptions(true);

31 // NOTE since get<T> this can cast int -> float or float -> int
32 // we need to check whether the underlying token type actually matches

35 if (entry.stream().tokens().size() == 1)
36 {
37 bool isLabel = entry.stream().tokens()[0].type() == Foam::token::tokenType::LABEL;
38 // we are testing whether the entryType is a scalar but the underlying token is

40 if (isLabel)
