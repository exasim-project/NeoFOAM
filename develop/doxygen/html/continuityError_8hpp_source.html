28inline ContinuityErrors computeContinuityError(
29 const NeoN::finiteVolume::cellCentred::SurfaceField<NeoN::scalar>& phi,
30 const RunTime& rt
31)
32{
33 const NeoN::scalar dt = rt.dt;
34 const auto nCells = static_cast<NeoN::localIdx>(rt.nfMesh.nCells());
35
36 NeoN::dsl::Expression<NeoN::scalar> divExpr(rt.exec);
37 divExpr.addOperator(NeoN::dsl::exp::div(phi));
38 NeoN::Vector<NeoN::scalar> divPhi = divExpr.explicitOperation(nCells);
39
40 const auto divPhiView = divPhi.view();
41 const auto volView = rt.nfMesh.cellVolumes().view();
42
43 NeoN::scalar localAbsVolSum = 0.0;
44 NeoN::parallelReduce(
45 rt.exec,
46 {0, static_cast<size_t>(nCells)},
47 NEON_LAMBDA(const size_t i, NeoN::scalar& s) {
48 s += Kokkos::abs(divPhiView[i]) * volView[i];
49 },
50 localAbsVolSum
51 );
52
53 NeoN::scalar localSignedVolSum = 0.0;
54 NeoN::parallelReduce(
55 rt.exec,
56 {0, static_cast<size_t>(nCells)},
57 NEON_LAMBDA(const size_t i, NeoN::scalar& s) { s += divPhiView[i] * volView[i]; },
58 localSignedVolSum
59 );
60
61 NeoN::scalar localTotalVol = 0.0;
62 NeoN::parallelReduce(
63 rt.exec,
64 {0, static_cast<size_t>(nCells)},
65 NEON_LAMBDA(const size_t i, NeoN::scalar& s) { s += volView[i]; },
66 localTotalVol
67 );
68
69 NeoN::scalar globalAbsVolSum = localAbsVolSum;
70 NeoN::scalar globalSignedVolSum = localSignedVolSum;
71 NeoN::scalar totalVol = localTotalVol;
72
73#ifdef NF_WITH_MPI_SUPPORT
74 if (rt.nfMesh.boundaryMesh().isDistributed())
75 {
76 NeoN::mpi::Environment env;
77 NeoN::scalar sums[3] = {globalAbsVolSum, globalSignedVolSum, totalVol};
78 MPI_Allreduce(
79 MPI_IN_PLACE,
80 sums,
81 3,
82 NeoN::mpi::getType<NeoN::scalar>(),
83 MPI_SUM,
84 env.comm()
85 );
86 globalAbsVolSum = sums[0];
87 globalSignedVolSum = sums[1];
88 totalVol = sums[2];
89 }
90#endif
91
92 return {dt * globalAbsVolSum / totalVol, dt * globalSignedVolSum / totalVol};
93}

102inline void reportContinuityError(
103 const NeoN::finiteVolume::cellCentred::SurfaceField<NeoN::scalar>& phi,
104 const RunTime& rt,
105 NeoN::scalar& cumulative

117}
