175 void setReference(NeoN::localIdx pRefCell, NeoN::scalar pRefValue)
176 {
177 // Accept the call in serial (MPI not initialised — mpiRank is the
178 // sentinel -1, which would compare unequal to 0 if cast to NeoN::sizeType)
179 // and on rank 0 in parallel. SetReference::operator() further guards
180 // the matrix mutation against non-rank-0 in initialised MPI.
181 // When runTime_ is null (1-arg constructor path), default to accepting
182 // on all ranks — SetReference::operator() guards non-rank-0 internally.
183 bool accept = true;
184 if (runTime_ != nullptr)
185 {
186 const auto& mpiEnv = runTime_->mpiEnvironment;
187 accept = !mpiEnv.isInitialized() || mpiEnv.rank() == 0;
188 }
189 if (accept)
190 {
191 needReference_ = true;
192 pRefCell_ = pRefCell;
193 pRefValue_ = pRefValue;
194 }

312 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
313 {
314 // Assemble and relax the owned ls_ so computeRAUandHByA reads the relaxed diagonal.
315 // Apply -grad p to ls_->rhs() in place (avoids copying the matrix), snapshot the rhs
316 // beforehand and restore it after solve so ls_ retains the H-system for computeRAUandHByA.
317 assemble();
318 relaxOwnedLs();
319
320
321 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
322 rhsExpr.read(NeoFOAM::expandSchemeDefaults(runTime_->fvSchemesDict, rhsExpr, psi_->name));
323 auto savedRhs = NeoN::Vector<ValueType>(ls_->rhs());
324 rhsExpr.assembleExplicitSource(*ls_, psi_->mesh());
325
326 auto solverDict = runTime_->fvSolutionDict.subDict("solvers");
327 const std::string finalKey = psi_->name + "Final";
328 const bool useFinal = finalIter_ && NeoFOAM::hasSolverSettings(solverDict, finalKey);
329 auto fvSolution = useFinal ? NeoFOAM::solverSettings(solverDict, finalKey)
330 : NeoFOAM::solverSettings(solverDict, psi_->name);
331 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
332 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
333 stripNeoFOAMKeys(fvSolution);
334 // Persist the solver across iterations, as solveImpl does, so its setup and caches survive.
335 const std::string solverKey = "solver:" + (useFinal ? finalKey : psi_->name);
336 auto& solver = readOrCreate<std::shared_ptr<NeoN::la::Solver>>(
337 *runTime_,
338 solverKey,
339 [&] { return std::make_shared<NeoN::la::Solver>(psi_->exec(), fvSolution); }
340 );
341 // Do some sanity checks before trying to solve
342 // NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
343 auto stats = solver->solve(*ls_, psi_->internalVector());
344
345 ls_->rhs() = savedRhs;
346
347 reportSolverStats(stats, fvSolution);
348 return stats;
349 }

352 NeoN::la::SolverStats solveWith(NeoN::la::Solver& solver, VolumeField& psi, RunTime& rt)
353 {
354 initIfNeeded(psi, rt);
355
356 expr_.read(NeoFOAM::expandSchemeDefaults(rt.fvSchemesDict, expr_, psi.name));
357 ls_->reset();
358 expr_.assemble(rt.t, rt.dt, *ls_, psi.mesh());
359
360 applyRelaxation(*ls_, psi, rt);
361
362 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
363 {
364 if (needReference_)
365 {
366 NeoN::dsl::SetReference<ValueType> refFunct(pRefCell_, pRefValue_);
367 refFunct(*ls_);
368 }
369 }
370
371 auto solverDict = rt.fvSolutionDict.subDict("solvers");
372 const std::string finalKey = psi.name + "Final";
373 const bool useFinal = finalIter_ && NeoFOAM::hasSolverSettings(solverDict, finalKey);
374 auto fieldSolverDict = useFinal ? NeoFOAM::solverSettings(solverDict, finalKey)
375 : NeoFOAM::solverSettings(solverDict, psi.name);
376 stripNeoFOAMKeys(fieldSolverDict);
377 NeoN::fence(psi.exec());
378 NF_ASSERT(ls_->exec() == psi.exec(), "Executors are not the same");
379 NeoN::la::SolverStats stats;
380 if (useFinal)
381 {
382 // The caller caches only the non-final solver; keep the final one in the RunTime like
383 // solveImpl, so it is not rebuilt in the final iteration of every time step.
384 auto& finalSolver = readOrCreate<std::shared_ptr<NeoN::la::Solver>>(
385 rt,
386 "solver:" + finalKey,
387 [&] { return std::make_shared<NeoN::la::Solver>(psi.exec(), fieldSolverDict); }
388 );
389 stats = finalSolver->solve(*ls_, psi.internalVector());
390 }
391 else
392 {
393 stats = solver.solve(*ls_, psi.internalVector());
394 }
395
396 reportSolverStats(stats, fieldSolverDict);
397 return stats;
398 }

453 // Public because NVCC forbids extended __host__ __device__ lambdas
454 // (NEON_LAMBDA) inside private or protected member functions.
455 NeoN::la::SolverStats solveImpl(dsl::Expression<ValueType>& expr, LinearSystem& ls)
456 {
457 // Re-read schemes (idempotent with the constructor read)
458 expr.read(NeoFOAM::expandSchemeDefaults(runTime_->fvSchemesDict, expr, psi_->name));
459
460 // Assemble without post-assembly functors; we apply SetReference separately below
461 // to ensure correct polymorphic dispatch — storing PostAssemblyBase by value causes
462 // object slicing that silently disables virtual overrides.
463 expr.assemble(runTime_->t, runTime_->dt, ls, psi_->mesh());
464
465 // Relaxation MUST precede SetReference: SetReference doubles the ref-cell diagonal,
466 // and relaxing afterwards would corrupt the pin.
467 applyRelaxation(ls, *psi_, *runTime_);
468
469 // Apply reference-cell pinning (kernel compiled only in pde.cpp via detail helper)
470 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
471 {
472 if (needReference_)
473 {
474 detail::applySetReference(ls, pRefCell_, pRefValue_);
475 }
476 }
477
478 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
479 {
480 if (constraintMask_ != nullptr)
481 {
482 const NeoN::localIdx nCells = psi_->mesh().nCells();
483 detail::applyFixedValueConstraints(
484 ls,
485 constraintMask_->view(),
486 constraintValues_->view(),
487 nCells
488 );
489 // Assign the field too, as fvMatrix::setValuesFromList does: the decoupled row
490 // only yields the pinned value once the solver has actually worked on it.
491 detail::assignFixedValues(
492 psi_->internalVector(),
493 constraintMask_->view(),
494 constraintValues_->view(),
495 nCells
496 );
497 }
498 }
499
500 auto solverDict = runTime_->fvSolutionDict.subDict("solvers");
501 const std::string finalKey = psi_->name + "Final";
502 const bool useFinal = finalIter_ && NeoFOAM::hasSolverSettings(solverDict, finalKey);
503 auto fieldSolverDict = useFinal ? NeoFOAM::solverSettings(solverDict, finalKey)
504 : NeoFOAM::solverSettings(solverDict, psi_->name);
505 // Drop NeoFOAM-only keys before handing the dict to NeoN/Ginkgo, whose
506 // config parser rejects unknown keys (e.g. assemblyStrategy, optimize).
507 stripNeoFOAMKeys(fieldSolverDict);
508 NeoN::fence(psi_->exec());
509 NF_ASSERT(ls.exec() == psi_->exec(), "Executors are not the same");
510
511 // Persist the NeoN solver across iterations: keep the solver instance alive in the
512 // RunTime controlDict (like ls_), keyed by field and its Final variant. This lets a
513 // Ginkgo solver reuse setup it keeps across solves (e.g. the multigrid hierarchy with
514 // reuseSetup) instead of rebuilding it every time -- a fresh per-solve solver would
515 // discard it. The initializer runs once, on the first solve for this key.
516 //
517 // Stored as a shared_ptr, NOT a bare Solver: Dictionary::insert copies the std::any, and a
518 // Solver copy invokes NeoN::la::Solver's copy ctor (solverInstance_->clone(), which the
519 // Ginkgo backend does not support). Copying the shared_ptr only bumps a refcount, so the
520 // Solver itself -- and its cached setup -- is never cloned.
521 const std::string solverKey = "solver:" + (useFinal ? finalKey : psi_->name);
522 auto& solver = readOrCreate<std::shared_ptr<NeoN::la::Solver>>(
523 *runTime_,
524 solverKey,
525 [&] { return std::make_shared<NeoN::la::Solver>(psi_->exec(), fieldSolverDict); }
526 );
527 auto stats = solver->solve(ls, psi_->internalVector());
528
529 reportSolverStats(stats, fieldSolverDict);
530
531 return stats;
532 }

534 // Relax the owned ls_ in place. Shared by solve(rhs) and assembleAndRelax() so both
535 // paths use the same alpha lookup. Must run before SetReference (see solveImpl).
536 // Only calls the free function applyMatrixRelaxation — no NEON_LAMBDA in this body.
537 void relaxOwnedLs() { applyRelaxation(*ls_, *psi_, *runTime_); }
538
539 // The single place equation relaxation is applied. Relaxation is a no-op unless relax() was
540 // called (fvMatrix::relax() semantics — see relax()); the post-relaxation constraint, if
541 // registered, runs either way (fvOptions.constrain does not depend on relax()).
542 void applyRelaxation(LinearSystem& ls, VolumeField& psi, const RunTime& rt)
543 {
544 if (relax_)
545 {
546 const auto alpha =
547 lookupEqnRelaxation(rt.fvSolutionDict, psi.name, finalIter_).value_or(1.0);
548 NeoN::dsl::applyMatrixRelaxation(ls, psi, alpha);
549 }
550 if (postRelaxConstraint_) postRelaxConstraint_(ls);
551 }

553 // Remove NeoFOAM-specific control keys from a per-field solver dict before it
554 // is passed to NeoN/Ginkgo. Ginkgo's config parser is strict and aborts on any
555 // unrecognised key, so these must be popped here rather than left in place.
556 static void stripNeoFOAMKeys(NeoN::Dictionary& dict)
557 {
558 for (const auto& key : {"assemblyStrategy", "optimize"})
559 {
560 if (dict.contains(key))
561 {
562 dict.remove(key);
563 }
564 }
565 }

609 // Per-component name (Ux/Uy/Uz) when a vector field is solved as separate
610 // component systems; the plain field name otherwise (scalar or coupled solve).
611 static std::string componentName(const std::string& base, std::size_t i, std::size_t n)
612 {
613 if (n <= 1) return base;
614 constexpr const char* suffix[3] = {"x", "y", "z"};
615 return i < 3 ? base + suffix[i] : base + "[" + std::to_string(i) + "]";
616 }

618 // OpenFOAM-style per-solve residual report: "<precond><solver>: Solving for
619 // <field>, Initial residual = ..., Final residual = ..., No Iterations N".
620 // The "<precond><solver>" label (e.g. DICPCG) is read from the solver dict's
621 // reportName meta key stashed by mapFvSolution -> no recomputation per solve.
622 // Logging is rank-aware and skips formatting on muted ranks (NeoN shouldLog).
623 void reportSolverStats(
624 const NeoN::la::SolverStats& stats,
625 const NeoN::Dictionary& fieldSolverDict
626 ) const
627 {
628 const std::string label = fieldSolverDict.contains("reportName")
629 ? fieldSolverDict.get<std::string>("reportName")
630 : std::string("Ginkgo");
631 const std::size_t n = stats.entries.size();
632 for (std::size_t i = 0; i < n; ++i)
633 {
634 const auto& stat = stats.entries[i];
635 NeoN::Logging::info(
636 "{}: Solving for {}, Initial residual = {}, Final residual = {}, No Iterations {}",
637 label,
638 componentName(psi_->name, i, n),
639 stat.initResNorm,
640 stat.finalResNorm,
641 stat.numIter
642 );
643 }
644 }

647 dsl::Expression<ValueType> expr_;
648 RunTime* runTime_;
649 // The linear system stored in the RunTime controlDict under "linearSystem<field>". Held by
650 // pointer, not copied: a PDE is typically constructed every time step, and a copy would
651 // reallocate the matrix each step, defeating the solver's caches keyed on its storage.
652 LinearSystem* ls_;
653 bool needReference_ = false;
654 NeoN::localIdx pRefCell_ = 0;
655 NeoN::scalar pRefValue_ = 0.0;
656 bool finalIter_ = false;
657 bool relax_ = false;
658 const NeoN::Vector<NeoN::scalar>* constraintMask_ = nullptr;
659 const NeoN::Vector<ValueType>* constraintValues_ = nullptr;
660 // Optional PDE-owned storage backing constraintMask_/constraintValues_ (setConstraintsOwned).
661 std::optional<NeoN::Vector<NeoN::scalar>> ownedConstraintMask_;
662 std::optional<NeoN::Vector<ValueType>> ownedConstraintValues_;
663 // Post-relaxation constraint on the assembled system (fvOptions.constrain analogue).
664 // Empty unless setPostRelaxConstraint() was called. See applyRelaxation().
665 std::function<void(LinearSystem&)> postRelaxConstraint_ {};

672template<typename ValueType, typename IndexType = NeoN::localIdx>
673NeoN::finiteVolume::cellCentred::VolumeField<ValueType> applyOperator(
674 const la::LinearSystem<ValueType>& ls,
675 const NeoN::finiteVolume::cellCentred::VolumeField<ValueType>& psi
676)
677{
678 NeoN::finiteVolume::cellCentred::VolumeField<ValueType> res(
679 psi.exec(),
680 "ls_" + psi.name,
681 psi.mesh(),
682 psi.internalVector(),
683 psi.boundaryData(),
684 psi.boundaryConditions()
685 );
686 NeoN::la::computeResidual(ls.matrix(), ls.rhs(), psi.internalVector(), res.internalVector());
687 return res;
688}
