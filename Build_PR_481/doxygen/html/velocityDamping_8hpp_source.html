50template<typename LinearSystemType, typename VolumeFieldType>
51NeoN::label applyVelocityDamping(
52 LinearSystemType& ls,
53 const VolumeFieldType& U,
54 NeoN::scalar UMax,
55 NeoN::scalar C = 1.0
56)
57{
58 // The segregated momentum system (PDE<Vec3>) has scalar coefficients; a coupled Vec3
59 // matrix would need the boost on every diagonal component.
60 static_assert(
61 std::is_same_v<
62 std::remove_cv_t<std::remove_reference_t<decltype(ls.matrix().values().view()[0])>>,
63 NeoN::scalar>,
64 "applyVelocityDamping expects a scalar-coefficient (segregated) linear system"
65 );
66
67 auto lsView = ls.view();
68 auto& matrix = lsView.matrix;
69 // Same addressing the relaxation kernel uses: diagIdx(celli) maps a cell to
70 // its diagonal slot in the CSR values array.
71 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
72 const auto uV = U.internalVector().view();
73 const auto volV = U.mesh().cellVolumes().view();
74
75 const auto nCells = static_cast<NeoN::localIdx>(uV.size());
76
77 NeoN::label nDamped = 0;
78 NeoN::parallelReduce(
79 U.exec(),
80 {0, nCells},
81 NEON_LAMBDA(const NeoN::localIdx celli, NeoN::label& count) {
82 const NeoN::scalar magU = NeoN::mag(uV[celli]);
83 if (magU > UMax)
84 {
85 // scale = sqr(cbrt(V)) -- V^(2/3), not V.
86 const NeoN::scalar cbrtV = Kokkos::cbrt(volV[celli]);
87 matrix.values[ma.diagIdx(celli)] += C * cbrtV * cbrtV * (magU - UMax);
88 ++count;
89 }
90 },
91 nDamped
92 );
93
94 return nDamped;
95}
