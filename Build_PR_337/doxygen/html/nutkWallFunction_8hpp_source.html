36inline void setNutkWallFunction(
37 Field<scalar>& domainVector,
38 const fvcc::VolumeField<scalar>& k,
39 const fvcc::VolumeField<scalar>& nu,
40 const fvcc::VolumeField<scalar>& nearWallDist,
41 const UnstructuredMesh& mesh,
42 std::pair<localIdx, localIdx> range,
43 scalar Cmu,
44 scalar kappa,
45 scalar E
46)
47{
48 const scalar Cmu25 = Kokkos::pow(Cmu, scalar(0.25));
49
50 const auto kInternal = k.internalVector().view();
51 const auto nuBoundary = nu.boundaryData().value().view();
52 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
53
54 auto [refGradient, value, valueFraction, refValue, faceCells] = views(
55 domainVector.boundaryData().refGrad(),
56 domainVector.boundaryData().value(),
57 domainVector.boundaryData().valueFraction(),
58 domainVector.boundaryData().refValue(),
59 mesh.boundaryMesh().faceOwners()
60 );
61
62 NeoN::parallelFor(
63 domainVector.exec(),
64 range,
65 NEON_LAMBDA(const localIdx i) {
66 const localIdx owner = faceCells[i];
67 const scalar y = nearWallBoundary[i];
68 const scalar nuw = nuBoundary[i];
69 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
70
71 const scalar yPlus = Cmu25 * y * Kokkos::sqrt(kw) / nuw;
72 const scalar nutLog =
73 nuw * yPlus * kappa / Kokkos::log(Kokkos::max(E * yPlus, scalar(1) + scalar(1e-4)));
74 const scalar nutw = Kokkos::sqrt(nuw * nuw + nutLog * nutLog);
75
76 value[i] = nutw;
77 refValue[i] = nutw;
78 valueFraction[i] = 1.0;
79 refGradient[i] = 0.0;
80 },
81 "setNutkWallFunction"
82 );
83}
