34inline void setEpsilonWallFunction(
35 Field<scalar>& epsilon,
36 const fvcc::VolumeField<scalar>& k,
37 const fvcc::VolumeField<scalar>& nu,
38 const fvcc::VolumeField<scalar>& nearWallDist,
39 const UnstructuredMesh& mesh,
40 std::pair<localIdx, localIdx> range,
41 scalar Cmu,
42 scalar kappa
43)
44{
45 const scalar Cmu75 = Kokkos::pow(Cmu, scalar(0.75));
46
47 auto kInternal = k.internalVector().view();
48 const auto nuBoundary = nu.boundaryData().value().view();
49 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
50
51 auto [refGrad, value, valueFraction, refValue, faceOwners] = views(
52 epsilon.boundaryData().refGrad(),
53 epsilon.boundaryData().value(),
54 epsilon.boundaryData().valueFraction(),
55 epsilon.boundaryData().refValue(),
56 mesh.boundaryMesh().faceOwners()
57 );
58
59 NeoN::parallelFor(
60 epsilon.exec(),
61 range,
62 NEON_LAMBDA(const localIdx i) {
63 const localIdx owner = faceOwners[i];
64 const scalar y = nearWallBoundary[i];
65 const scalar nuw = nuBoundary[i];
66 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
67
68 const scalar eVis = scalar(2) * kw * nuw / (y * y);
69 const scalar eLog = Cmu75 * Kokkos::pow(kw, scalar(1.5)) / (kappa * y);
70 const scalar eOmega = Kokkos::sqrt(eVis * eVis + eLog * eLog);
71
72 value[i] = eOmega;
73 refValue[i] = eOmega;
74 valueFraction[i] = 1.0;
75 refGrad[i] = 0.0;
76 },
77 "setEpsilonWallFunction"
78 );
79}
