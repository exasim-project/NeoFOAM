15// Mirrors OpenFOAM's kqRWallFunctionFvPatchField<scalar>
16// (src/TurbulenceModels/turbulenceModels/derivedFvPatchFields/wallFunctions/
17// kqRWallFunctions/kqRWallFunction/kqRWallFunctionFvPatchField.{H,C}).
18//
19// Upstream the class derives from zeroGradientFvPatchField<Type> and overrides
20// nothing — its own docstring states "It is not a wall-function condition.".
21// We mirror that here: correctBoundaryCondition() applies the same kernel as
22// FixedGradient<scalar> with a zero gradient.
23//
24// Kept as a distinct dispatched class (not a string-level alias of
25// fixedGradient) so OpenFOAM dictionaries authored with `type kqRWallFunction;`
26// flow through the runtime-selection factory unchanged. This also leaves the
27// door open for future wall functions (omegaWallFunction, nutkWallFunction)
28// that *do* need a real wall-distance-based kernel — they can be registered
29// alongside this class and consume BoundaryContext::scalarFieldPtr("nearWallDist")
30// the way NutUSpaldingWallFunction already does.
31class KqRWallFunction : public VolumeBoundaryFactory<scalar>::template Register<KqRWallFunction>
32{
33 using Base = VolumeBoundaryFactory<scalar>::template Register<KqRWallFunction>;
34
35public:
36
37 using Base::correctBoundaryCondition;
38
39 KqRWallFunction(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
40 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = false})
41 , mesh_(mesh)
42 {}
43
44 void correctBoundaryCondition(Field<scalar>& domainVector) final
45 {
46 detail::setGradientValue(domainVector, mesh_, this->range(), scalar(0));
47 }
48
49 static std::string name() { return "kqRWallFunction"; }
50
51 std::string getName() const override { return name(); }
52
53 static std::string doc()
54 {
55 return "Renamed zero-gradient BC for turbulent kinetic energy k (and q, R) "
56 "at high-Re walls. Matches OpenFOAM's kqRWallFunction, which is not "
57 "actually a wall function — see upstream class docstring.";
58 }
59
60 static std::string schema() { return "none"; }
61
62 std::unique_ptr<VolumeBoundaryFactory<scalar>> clone() const final
63 {
64 return std::make_unique<KqRWallFunction>(*this);
65 }
66
67private:
68
69 const UnstructuredMesh& mesh_;
70};
