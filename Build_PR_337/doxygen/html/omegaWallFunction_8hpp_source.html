40inline void setOmegaWallFunction(
41 Field<scalar>& omega,
42 const fvcc::VolumeField<scalar>& k,
43 const fvcc::VolumeField<scalar>& nu,
44 const fvcc::VolumeField<scalar>& nearWallDist,
45 const UnstructuredMesh& mesh,
46 std::pair<localIdx, localIdx> range,
47 scalar beta1,
48 scalar Cmu,
49 scalar kappa
50)
51{
52 const scalar Cmu25 = Kokkos::pow(Cmu, scalar(0.25));
53
54 auto kInternal = k.internalVector().view();
55 const auto nuBoundary = nu.boundaryData().value().view();
56 const auto nearWallBoundary = nearWallDist.boundaryData().value().view();
57
58 auto [refGrad, value, valueFraction, refValue, faceOwners] = views(
59 omega.boundaryData().refGrad(),
60 omega.boundaryData().value(),
61 omega.boundaryData().valueFraction(),
62 omega.boundaryData().refValue(),
63 mesh.boundaryMesh().faceOwners()
64 );
65
66 NeoN::parallelFor(
67 omega.exec(),
68 range,
69 NEON_LAMBDA(const localIdx i) {
70 const localIdx owner = faceOwners[i];
71 const scalar y = nearWallBoundary[i];
72 const scalar nuw = nuBoundary[i];
73 const scalar kw = Kokkos::max(kInternal[owner], scalar(0));
74
75 const scalar wVis = 6.0 * nuw / (beta1 * y * y);
76 const scalar wLog = Kokkos::sqrt(kw) / (Cmu25 * kappa * y);
77 const scalar wOmega = Kokkos::sqrt(wVis * wVis + wLog * wLog);
78
79 value[i] = wOmega;
80 refValue[i] = wOmega;
81 valueFraction[i] = 1.0;
82 refGrad[i] = 0.0;
83 },
84 "setOmegaWallFunction"
85 );
86}
