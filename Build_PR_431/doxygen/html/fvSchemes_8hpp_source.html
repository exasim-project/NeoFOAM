54template<typename ValueType>
55NeoN::Dictionary expandSchemeDefaults(
56 const NeoN::Dictionary& schemesDict,
57 const NeoN::dsl::Expression<ValueType>& expr,
58 const std::string& fieldName
59)
60{
61 NeoN::Dictionary resolved = schemesDict;
62
63 auto fillDefault = [&](const std::string& subName, const std::string& key)
64 {
65 if (!resolved.contains(subName)) return;
66 auto& sub = resolved.subDict(subName);
67 if (!sub.contains(key) && sub.contains("default"))
68 {
69 if (sub.template isType<std::string>("default"))
70 {
71 const auto& defaultStr = sub.get<std::string>("default");
72 if (defaultStr == "none")
73 return; // "none" means no fallback — require explicit entry
74 NeoN::TokenList tl;
75 tl.insert(defaultStr);
76 sub.insert(key, tl);
77 }
78 else
79 sub.insert(key, sub.get<NeoN::TokenList>("default"));
80 }
81 };
82
83 // ddt: DdtOperator::read() expects std::string, not TokenList — insert directly
84 if (resolved.contains("ddtSchemes"))
85 {
86 auto& ddtSub = resolved.subDict("ddtSchemes");
87 const std::string ddtKey = "ddt(" + fieldName + ")";
88 if (!ddtSub.contains(ddtKey) && ddtSub.contains("default")
89 && ddtSub.template isType<std::string>("default"))
90 {
91 const auto& def = ddtSub.get<std::string>("default");
92 if (def != "none") ddtSub.insert(ddtKey, def);
93 }
94 }
95 fillDefault("gradSchemes", "grad(" + fieldName + ")");
96
97 // div and laplacian need the flux/gamma names from each operator's config
98 for (const auto& op : expr.spatialOperators())
99 {
100 if (op.getName() == "ViscousStressOperator")
101 {
102 fillDefault("divSchemes", "div((nuEff*dev2(T(grad(U)))))");
103 continue;
104 }
105
106 if (op.getName() == "GradOperator")
107 {
108 auto config = op.getConfig();
109 if (config.contains("field"))
110 {
111 const auto& scalarField =
112 config
113 .template get<NeoN::detail::RefHolder<fvcc::VolumeField<NeoN::scalar>>>(
114 "field"
115 )
116 .c;
117 fillDefault("gradSchemes", "grad(" + scalarField.name + ")");
118 }
119 continue;
120 }
121
122 auto config = op.getConfig();
123 if (!config.contains("field")) continue;
124
125 const auto& field =
126 config.template get<NeoN::detail::RefHolder<fvcc::VolumeField<ValueType>>>("field").c;
127
128 if (op.getName() == "DivOperator" && config.contains("flux"))
129 {
130 const auto& flux =
131 config
132 .template get<NeoN::detail::RefHolder<fvcc::SurfaceField<NeoN::scalar>>>("flux")
133 .c;
134 fillDefault("divSchemes", "div(" + flux.name + "," + field.name + ")");
135 }
136 else if (op.getName() == "LaplacianOperator" && config.contains("gamma"))
137 {
138 const auto& gamma =
139 config
140 .template get<NeoN::detail::RefHolder<fvcc::SurfaceField<NeoN::scalar>>>("gamma"
141 )
142 .c;
143 fillDefault("laplacianSchemes", "laplacian(" + gamma.name + "," + field.name + ")");
144 }
145 }
146
147 return resolved;
148}
