4// Per-rank proc-boundary value dumps for offline CPU-vs-GPU bisection.
5//
6// Gated entirely on env NEOFOAM_PROC_DUMP=1 — when unset, every call is a
7// no-op (single getenv at first call, cached). For the bisect workflow see
8// tools/hpc-bisect-gpu-proc-divergence.sh.
9//
10// Output one file per (step, pisoIter, nonOrthIter, checkpoint, fieldName):
11// processor{rank}/dumps/<step:04d>_<piso>_<nonOrth>_<checkpoint>__<field>.txt
12//
13// Each row is "<patch> <face> <value...>" with values formatted at %.10e so
14// CPU↔GPU FP roundoff (~1e-15) rounds to identical text and real divergence
15// (the bug we're hunting) is visible to strict diff(1).

78template<typename Field>
79void dumpProcFaces(
80 const Field& field,
81 const std::string& fieldName,
82 const std::string& checkpoint,
83 int stepIdx,
84 int pisoIter,
85 int nonOrthIter
86)
87{
88 if (!procDumpEnabled()) return;
89
90 const auto& bm = field.mesh().boundaryMesh();
91 if (!bm.isDistributed()) return;
92
93 const auto procPatchCount = bm.nProcBoundaryPatches();
94 if (procPatchCount == 0) return;
95
96 int rank = 0;
97 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
98
99 auto perPatchBoundary = detail::extractProcBoundaryValues(field);
100
101 namespace fs = std::filesystem;
102 const fs::path dir = fs::path("processor" + std::to_string(rank)) / "dumps";
103 std::error_code ec;
104 fs::create_directories(dir, ec);
105
106 char filename[512];
107 std::snprintf(
108 filename,
109 sizeof(filename),
110 "%s/%04d_%d_%d_%s__%s.txt",
111 dir.string().c_str(),
112 stepIdx,
113 pisoIter,
114 nonOrthIter,
115 checkpoint.c_str(),
116 fieldName.c_str()
117 );
118
119 std::FILE* f = std::fopen(filename, "w");
120 if (!f) return;
121
122 for (NeoN::localIdx p = 0; p < procPatchCount; ++p)
123 {
124 const auto nFaces = perPatchBoundary[p].size();
125 for (std::size_t bf = 0; bf < nFaces; ++bf)
126 {
127 detail::dumpRow(
128 f,
129 static_cast<int>(p),
130 static_cast<int>(bf),
131 perPatchBoundary[p][bf]
132 );
133 }
134 }
135 std::fclose(f);
136}

246// Single template overload handles all integer widths uniformly: localIdx
247// (== int on the default 32-bit-label build) and the int values inside
248// commPattern.sendCounts / recvIdx. Cast to long long so any signed integer
249// fits and the output format is stable across builds.
250template<typename Integral, std::enable_if_t<std::is_integral_v<Integral>, int> = 0>
251inline void dumpVectorRow(std::FILE* f, std::size_t i, Integral v)
252{
253 std::fprintf(f, "0 %zu %lld\n", i, static_cast<long long>(v));
254}

382template<typename LinearSystem, typename Vector>
383void dumpFullLinearSystem(
384 const LinearSystem& ls,
385 const Vector& x,
386 const std::string& systemName,
387 const std::string& checkpoint,
388 int stepIdx,
389 int pisoIter,
390 int nonOrthIter
391)
392{
393 if (!procDumpEnabled()) return;
394
395 // Local matrix
396 dumpVector(
397 ls.matrix().values(),
398 systemName + "_localA_values",
399 checkpoint,
400 stepIdx,
401 pisoIter,
402 nonOrthIter
403 );
404 dumpVector(
405 ls.matrix().sparsity()->colIdxs(),
406 systemName + "_localA_colIdxs",
407 checkpoint,
408 stepIdx,
409 pisoIter,
410 nonOrthIter
411 );
412 dumpVector(
413 ls.matrix().sparsity()->rowOffs(),
414 systemName + "_localA_rowOffs",
415 checkpoint,
416 stepIdx,
417 pisoIter,
418 nonOrthIter
419 );
420
421 // Non-local matrix
422 dumpVector(
423 ls.nonLocalMatrix().values(),
424 systemName + "_nonLocalA_values",
425 checkpoint,
426 stepIdx,
427 pisoIter,
428 nonOrthIter
429 );
430 dumpVector(
431 ls.nonLocalMatrix().sparsity()->colIdxs(),
432 systemName + "_nonLocalA_colIdxs",
433 checkpoint,
434 stepIdx,
435 pisoIter,
436 nonOrthIter
437 );
438 dumpVector(
439 ls.nonLocalMatrix().sparsity()->rowOffs(),
440 systemName + "_nonLocalA_rowOffs",
441 checkpoint,
442 stepIdx,
443 pisoIter,
444 nonOrthIter
445 );
446
447 // RHS and initial iterate
448 dumpVector(ls.rhs(), systemName + "_rhs", checkpoint, stepIdx, pisoIter, nonOrthIter);

451 // CommPattern (host-side integer arrays)

453 ls.commPattern().sendCounts,

455 checkpoint,
456 stepIdx,
457 pisoIter,
458 nonOrthIter
459 );
460 dumpHostVector(
