28inline bool valuesMatch(NeoN::scalar a, NeoN::scalar b, double tol)
29{
30 // host-side face comparison utility — std::abs/max are safe here (not device-compiled)
31 const double diff = std::abs(static_cast<double>(a) - static_cast<double>(b));
32 const double mag = std::max(std::abs(static_cast<double>(a)), std::abs(static_cast<double>(b)));
33 return diff <= tol * std::max(1.0, mag);
34}
35
36inline bool valuesMatch(NeoN::Vec3 a, NeoN::Vec3 b, double tol)
37{
38 return valuesMatch(a[0], b[0], tol) && valuesMatch(a[1], b[1], tol)
39 && valuesMatch(a[2], b[2], tol);
40}

83// Read the proc-tail of boundaryData().value() into a host vector keyed by
84// (procPatchIdx, faceWithinPatch).
85template<typename Field>
86auto extractProcBoundaryValues(const Field& field)
87{
88 using ValueType = typename Field::VectorValueType;
89 auto host = field.boundaryData().value().copyToHost();
90 const auto view = host.view();
91
92 const auto& bm = field.mesh().boundaryMesh();
93 const auto totalPatches = bm.nBoundaries();
94 const auto procPatchCount = bm.nProcBoundaryPatches();
95 const auto firstProcPatch = totalPatches - procPatchCount;
96 const auto& patchOffsets = bm.offset();
97
98 std::vector<std::vector<ValueType>> perPatchValues(procPatchCount);
99 for (NeoN::localIdx p = 0; p < procPatchCount; ++p)
100 {
101 const auto patchIdx = firstProcPatch + p;
102 const auto start = patchOffsets[patchIdx];
103 const auto end = patchOffsets[patchIdx + 1];
104 perPatchValues[p].reserve(end - start);
105 for (auto bf = start; bf < end; ++bf)
106 {
107 perPatchValues[p].push_back(view[bf]);
108 }
109 }
110 return perPatchValues;
111}

113// For a VolumeField: for each proc face, look up the LOCAL owner cell's
114// internal value (this is the "other rank's ghost" from the neighbour's side).
115template<typename Field>
116auto extractProcOwnerInternalValues(const Field& field)
117{
118 using ValueType = typename Field::VectorValueType;
119 auto hostInternal = field.internalVector().copyToHost();
120 const auto internalView = hostInternal.view();
121
122 const auto& bm = field.mesh().boundaryMesh();
123 auto faceCellsHost = bm.faceCells().copyToHost();
124 const auto faceCellsView = faceCellsHost.view();
125
126 const auto totalPatches = bm.nBoundaries();
127 const auto procPatchCount = bm.nProcBoundaryPatches();
128 const auto firstProcPatch = totalPatches - procPatchCount;
129 const auto& patchOffsets = bm.offset();
130 const auto nInternalFaces = field.mesh().nInternalFaces();
131
132 std::vector<std::vector<ValueType>> perPatchValues(procPatchCount);
133 for (NeoN::localIdx p = 0; p < procPatchCount; ++p)
134 {
135 const auto patchIdx = firstProcPatch + p;
136 const auto start = patchOffsets[patchIdx];
137 const auto end = patchOffsets[patchIdx + 1];
138 perPatchValues[p].reserve(end - start);
139 for (auto bf = start; bf < end; ++bf)
140 {
141 // bf indexes into boundaryData / faceCells (compressed boundary tail).
142 const auto ownerCell = faceCellsView[bf];
143 perPatchValues[p].push_back(internalView[ownerCell]);
144 }
145 }
146 return perPatchValues;
147}

149template<typename Field>
150bool fieldIsVolume(const Field& field)
151{
152 // VolumeField: internalVector size == nCells.
153 // SurfaceField: internalVector size == nTotalFaces.
154 return field.internalVector().size() == field.mesh().nCells();

205template<typename Field>
206bool checkProcFaceConsistency(
207 const Field& field,
208 const std::string& label,
209 double tolerance = 1e-12,
210 SignConvention sign = SignConvention::Equal
211)
212{
213 using ValueType = typename Field::VectorValueType;
214
215 const auto& bm = field.mesh().boundaryMesh();
216 if (!bm.isDistributed())
217 {
218 return true;
219 }
220
221 int myRank = 0;
222 MPI_Comm_rank(MPI_COMM_WORLD, &myRank);
223
224 const auto procPatchCount = bm.nProcBoundaryPatches();
225 const auto& nbrRanks = bm.neighbourRank();
226
227 const bool isVolume = detail::fieldIsVolume(field);
228
229 // Local boundaryData (proc-tail) — what we will assert.
230 auto perPatchBoundary = detail::extractProcBoundaryValues(field);
231 // Local owner-cell internal value at each proc face — what we send for
232 // VolumeFields. Only used in the volume case.
233 auto perPatchOwner = isVolume
234 ? detail::extractProcOwnerInternalValues(field)
235 : std::vector<std::vector<ValueType>> {};
236
237 bool allMatch = true;
238 for (NeoN::localIdx p = 0; p < procPatchCount; ++p)
239 {
240 const int nbrRank = static_cast<int>(nbrRanks[p]);
241 const int nFaces = static_cast<int>(perPatchBoundary[p].size());
242
243 // Pick what to send: owner-cell internal for VolumeField, boundaryData
244 // for SurfaceField.
245 const std::vector<ValueType>& sendBuf =
246 isVolume ? perPatchOwner[p] : perPatchBoundary[p];
247
248 std::vector<ValueType> remote(nFaces);
249 const int countMul = detail::MpiTraits<ValueType>::countMultiplier;
250
251 MPI_Sendrecv(
252 sendBuf.data(),
253 nFaces * countMul,
254 detail::MpiTraits<ValueType>::type(),
255 nbrRank,
256 7777,
257 remote.data(),
258 nFaces * countMul,
259 detail::MpiTraits<ValueType>::type(),
260 nbrRank,
261 7777,
262 MPI_COMM_WORLD,
263 MPI_STATUS_IGNORE
264 );
265
266 for (int f = 0; f < nFaces; ++f)
267 {
268 // Sign-flip applies only to surface flux fields. VolumeField
269 // ghost values are scalars/vectors with no flip.
270 ValueType expected =
271 (!isVolume && sign == SignConvention::FlipExpected)
272 ? detail::negate(remote[f])
273 : remote[f];
274 // Compare against local boundaryData regardless of field type:
275 // that's the value supposed to mirror the neighbour.
276 const ValueType& localVal = perPatchBoundary[p][f];
277 if (!detail::valuesMatch(localVal, expected, tolerance))
278 {
279 allMatch = false;
280 std::fprintf(
281 stderr,
282 "[procFaceCheck] %s: rank %d patch %d (-> %d) face %d MISMATCH "
283 "local.boundary=%s remote%s=%s\n",
284 label.c_str(),
285 myRank,
286 static_cast<int>(p),
287 nbrRank,
288 f,
289 detail::formatValue(localVal).c_str(),
290 isVolume ? ".internal[ownerOfFace]" : ".boundary",
291 detail::formatValue(remote[f]).c_str()
292 );
293 }
294 }
295 }
296
297 // Reduce so every rank knows the global verdict; useful for one-line
298 // pass/fail logs.
299 int localOk = allMatch ? 1 : 0;
300 int globalOk = 0;
301 MPI_Allreduce(&localOk, &globalOk, 1, MPI_INT, MPI_LAND, MPI_COMM_WORLD);
302
303 if (myRank == 0)
304 {
305 if (globalOk)
306 {
307 std::fprintf(stderr, "[procFaceCheck] %s: OK\n", label.c_str());
308 }
309 else
310 {
311 std::fprintf(stderr, "[procFaceCheck] %s: FAILED (see mismatches above)\n", label.c_str());
312 }
313 }
314 return globalOk != 0;
315}
