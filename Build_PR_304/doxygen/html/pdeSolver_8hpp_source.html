80 void setReference(NeoN::localIdx pRefCell, NeoN::scalar pRefValue)
81 {
82 // Accept the call in serial (MPI not initialised — mpiRank is the
83 // sentinel -1, which would compare unequal to 0 if cast to size_t)
84 // and on rank 0 in parallel. SetReference::operator() further guards
85 // the matrix mutation against non-rank-0 in initialised MPI.
86 const auto& mpiEnv = runTime_.mpiEnvironment;
87 if (!mpiEnv.isInitialized() || mpiEnv.rank() == 0)

110 NeoN::la::LinearSystem<ValueType> assemble(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
111 {
112 auto rhsExpr = dsl::Expression<ValueType>(-1.0 * rhs);
113 rhsExpr.read(runTime_.fvSchemesDict);
114 auto ls = NeoN::la::LinearSystem<ValueType>(assemble());
115
116 // Collect explicit contributions from the main expression first, then from the extra rhs.
117 // The main expression's explicit ops are subtracted from rhs (DSL convention), so they
118 // must be included here just as iterativeSolveImpl does for the no-rhs overload.
119 auto expTmp = expr_.explicitOperation(psi_.mesh().nCells());
120 rhsExpr.explicitOperation(expTmp);
121
122 auto [vol, expSource, rhsV] = NeoN::views(psi_.mesh().cellVolumes(), expTmp, ls.rhs());
123 NeoN::parallelFor(
124 psi_.exec(),
125 {0, rhsV.size()},
126 NEON_LAMBDA(const NeoN::localIdx i) { rhsV[i] -= expSource[i] * vol[i]; }
127 );
128
129 return ls;
130 }

144 NeoN::la::SolverStats solve(dsl::SpatialOperator<NeoN::Vec3>&& rhs)
145 {
146 // assemble wo rhs first
147 auto ls = assemble(std::move(rhs));
148
149 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
150 auto fvSolution = solverDict.subDict(psi_.name);
151 auto solver = NeoN::la::Solver(psi_.exec(), fvSolution);
152 // Do some sanity checks before trying to solve
153 // NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
154 auto stats = solver.solve(ls, psi_.internalVector());
155
156 for (auto& stat : stats.entries)
157 {
158 NeoN::Logging::info(
159 "Solving for {} Initial residual: {} Final residual: {} No Iterations: {}",
160 psi_.name,
161 stat.initResNorm,
162 stat.finalResNorm,
163 stat.numIter
164 );
165 }
166 return stats;
167 }

169 // Public because NVCC forbids extended __host__ __device__ lambdas
170 // (NEON_LAMBDA) inside private or protected member functions.
171 NeoN::la::SolverStats solveImpl(dsl::Expression<ValueType>& expr, LinearSystem& ls)
172 {
173 // Re-read schemes (idempotent with the constructor read)
174 expr.read(runTime_.fvSchemesDict);
175
176 // Assemble without post-assembly functors; we apply SetReference separately below
177 // to ensure correct polymorphic dispatch — storing PostAssemblyBase by value causes
178 // object slicing that silently disables virtual overrides.
179 expr.assemble(runTime_.t, runTime_.dt, ls);
180
181 // Subtract the explicit source term from the rhs (mirrors iterativeSolveImpl)
182 auto expTmp = expr.explicitOperation(psi_.mesh().nCells());
183 auto [vol, expSource, rhs] = NeoN::views(psi_.mesh().cellVolumes(), expTmp, ls.rhs());
184 NeoN::parallelFor(
185 psi_.exec(),
186 {0, static_cast<NeoN::localIdx>(rhs.size())},
187 NEON_LAMBDA(const NeoN::localIdx i) { rhs[i] -= expSource[i] * vol[i]; }
188 );
189
190 // Apply reference-cell pinning directly (avoids object-slicing issue)
191 if constexpr (std::is_same_v<ValueType, NeoN::scalar>)
192 {
193 if (needReference_)
194 {
195 NeoN::dsl::SetReference<ValueType> refFunct(pRefCell_, pRefValue_);
196 refFunct(ls);
197 }
198 }
199
200 auto solverDict = runTime_.fvSolutionDict.subDict("solvers");
201 auto fieldSolverDict = solverDict.subDict(psi_.name);
202 NeoN::fence(psi_.exec());
203 NF_ASSERT(ls.exec() == psi_.exec(), "Executors are not the same");
204
205 auto solver = NeoN::la::Solver(psi_.exec(), fieldSolverDict);
206 auto stats = solver.solve(ls, psi_.internalVector());
207
208 for (auto& stat : stats.entries)
209 {
210 NeoN::Logging::info(
211 "Solving for {} Initial residual: {} Final residual: {} No Iterations: {}",
212 psi_.name,
213 stat.initResNorm,
214 stat.finalResNorm,
215 stat.numIter
216 );
217 }
218
219 return stats;
220 }

234template<typename ValueType, typename IndexType = NeoN::localIdx>
235NeoN::finiteVolume::cellCentred::VolumeField<ValueType> applyOperator(
236 const la::LinearSystem<ValueType>& ls,
237 const NeoN::finiteVolume::cellCentred::VolumeField<ValueType>& psi
238)
239{
240 NeoN::finiteVolume::cellCentred::VolumeField<ValueType> res(
241 psi.exec(),
242 "ls_" + psi.name,
243 psi.mesh(),
244 psi.internalVector(),
245 psi.boundaryData(),
246 psi.boundaryConditions()
247 );
248 NeoN::la::computeResidual(ls.matrix(), ls.rhs(), psi.internalVector(), res.internalVector());
249 return res;
250}
